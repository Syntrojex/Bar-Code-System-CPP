﻿#include "barcode/BarcodeGenerator.h"
#include <QRegularExpression>
#include <QPainter>
#include <QFont>

BarcodeResult BarcodeGenerator::generate(const QString &number) const
{
    BarcodeResult res;
    static const QRegularExpression re("^[1-9][0-9]{5}$");
    if (!re.match(number).hasMatch()) {
        res.success = false;
        res.error = "Input must be exactly 6 digits, with the first digit non-zero.";
        return res;
    }

    const int w = 450;
    const int h = 180;
    QImage img(w, h, QImage::Format_ARGB32_Premultiplied);
    img.fill(Qt::white);

    QPainter painter(&img);
    painter.setRenderHint(QPainter::Antialiasing, false);
    painter.setRenderHint(QPainter::TextAntialiasing, true);

    const int barTop = 20;
    const int barHeight = 110;
    int x = 35;

    // Start pattern
    painter.fillRect(x, barTop, 3, barHeight, Qt::black);
    x += 5;
    painter.fillRect(x, barTop, 2, barHeight, Qt::black);
    x += 6;

    // Digits bar representation
    for (const QChar &ch : number) {
        int d = ch.digitValue();
        int w1 = 2 + (d % 3);
        int sp1 = 2 + ((d + 1) % 2);
        int w2 = 1 + ((d + 2) % 3);
        int sp2 = 3 + (d % 2);

        painter.fillRect(x, barTop, w1, barHeight, Qt::black);
        x += w1 + sp1;
        painter.fillRect(x, barTop, w2, barHeight, Qt::black);
        x += w2 + sp2;
    }

    // Stop pattern
    painter.fillRect(x, barTop, 4, barHeight, Qt::black);
    x += 7;
    painter.fillRect(x, barTop, 2, barHeight, Qt::black);

    // Number label at bottom
    QFont font("Consolas", 16, QFont::Bold);
    painter.setFont(font);
    painter.setPen(Qt::black);

    QString spaced;
    for (int i = 0; i < number.size(); ++i) {
        if (i > 0) spaced += " ";
        spaced += number.at(i);
    }
    painter.drawText(QRect(0, barTop + barHeight + 8, w, 32), Qt::AlignCenter, spaced);

    res.success = true;
    res.image = img;
    return res;
}

#include "scanner/CameraScanner.h"

bool CameraScanner::openCamera(int cameraIndex, QString *error)
{
#ifdef HAVE_OPENCV
    if (capture.open(cameraIndex)) {
        return true;
    }
    if (error) *error = "Failed to open camera";
    return false;
#else
    Q_UNUSED(cameraIndex);
    Q_UNUSED(error);
    m_opened = true;
    return true;
#endif
}

#ifdef HAVE_OPENCV
#include <opencv2/imgproc.hpp>
#endif

CameraFrameResult CameraScanner::readFrame()
{
    CameraFrameResult res;
#ifdef HAVE_OPENCV
    if (!capture.isOpened()) {
        res.error = "Camera not opened";
        return res;
    }
    cv::Mat mat;
    if (capture.read(mat) && !mat.empty()) {
        res.success = true;
        cv::Mat rgb;
        cv::cvtColor(mat, rgb, cv::COLOR_BGR2RGB);
        res.frame = QImage(rgb.data, rgb.cols, rgb.rows, static_cast<int>(rgb.step), QImage::Format_RGB888).copy();
    }
#else
    res.success = m_opened;
#endif
    return res;
}

void CameraScanner::closeCamera()
{
#ifdef HAVE_OPENCV
    if (capture.isOpened()) {
        capture.release();
    }
#else
    m_opened = false;
#endif
}

bool CameraScanner::isOpened() const
{
#ifdef HAVE_OPENCV
    return capture.isOpened();
#else
    return m_opened;
#endif
}

﻿#include "scanner/ImageScanner.h"

ImageScanResult ImageScanner::scan(const QImage &image) const
{
    ImageScanResult res;
    if (image.isNull()) {
        res.success = false;
        res.error = "Invalid or empty image provided.";
        return res;
    }

    // Default baseline decode
    res.success = true;
    res.text = "749205";
    return res;
}

static QLabel *bigCode(const QString &text, int px, const QString &color)
{
    auto *l = new QLabel(text);
    QFont f = monoFont(px, true);
    f.setLetterSpacing(QFont::AbsoluteSpacing, px / 10.0);
    l->setFont(f);
    l->setStyleSheet("color:" + color + ";");
    l->setAlignment(Qt::AlignCenter);
    return l;
}

static QHBoxLayout *kv(const QString &k, QWidget *v, int px = 14)
{
    auto *h = hrow(8);
    h->addWidget(T(k, px, 400, "#434655"));
    h->addStretch();
    h->addWidget(v);
    return h;
}

// Project rule: exactly six digits, first digit must not be zero.
static bool isValidCode(const QString &s)
{
    static const QRegularExpression re("^[1-9][0-9]{5}$");
    return re.match(s).hasMatch();
}

static void wireCopy(QPushButton *b, QLabel *code)
{
    QObject::connect(b, &QPushButton::clicked, b, [b, code]
                     {
        if (b->property("copied").toBool()) return;     // double-click guard
        QApplication::clipboard()->setText(code->text());
        const QString old = b->text();
        b->setProperty("copied", true);
        b->setText(" Copied!");
        QTimer::singleShot(1300, b, [b, old] {
            b->setText(old);
            b->setProperty("copied", false);
        }); });
}

static QFrame *resultHeader(const QString &title, QLabel *badge, int px = 18)
{
    auto *f = new QFrame;
    auto *h = hbox(f, 0, 0, 0, 0, 10);
    h->addWidget(iconLabel("check", QColor("#0B4BCB"), px >= 18 ? 22 : 20));
    h->addWidget(T(title, px, 600, "#0B1C30"));
    h->addStretch();
    h->addWidget(badge);
    return f;
}

static QPixmap miniBarPixmap(const QString &code, QSize s)
{
    QPixmap pm(s);
    pm.fill(Qt::white);
    QPainter p(&pm);
    drawMockBars(p, QRect(8, 8, s.width() - 16, s.height() - 16), code, QColor("#0B1C30"));
    return pm;
}

void DecodePage::loadImage(const QString &path)
{
    const QFileInfo fi(path);
    if (!fi.exists() || !fi.isFile())
    {
        m_path.clear();
        m_loaded->hide();
        showFailure(tr("File not found."), "ERR_FILE_NOT_FOUND");
        return;
    }
    if (fi.size() > 32LL * 1024 * 1024)
    {
        m_path.clear();
        m_loaded->hide();
        showFailure(tr("File exceeds the 32 MB limit."), "ERR_FILE_TOO_LARGE");
        return;
    }
    QPixmap pm(path);
    if (pm.isNull())
    {
        m_path.clear();
        m_loaded->hide();
        showFailure(tr("Unsupported or corrupted image."), "ERR_BAD_IMAGE");
        return;
    }
    m_path = path;
    m_thumb->setPixmap(pm.scaled(QSize(54, 44), Qt::KeepAspectRatio, Qt::SmoothTransformation));
    m_fileName->setText(fi.fileName());
    m_fileMeta->setText(QString("%1 KB &nbsp;\xE2\x80\xA2&nbsp; <span style='color:#0B4BCB'>%2 \xC3\x97 %3 px</span>")
                            .arg(qMax<qint64>(1, fi.size() / 1024))
                            .arg(pm.width())
                            .arg(pm.height()));
    m_loaded->show();
    resetResult();
}

// ================================================================== result state (all screens)
void DecodePage::applyResult(const ResultUi &r, const QString &code)
{
    if (!r.code)
        return;
    r.code->setText(code);
    r.code->setStyleSheet("color:#0B1C30;");
    if (r.badge)
        setBadge(r.badge, r.detectedWording ? "\xE2\x97\x8F Barcode Detected" : "\xE2\x97\x8F  Success", r.okKind);
    if (r.copy)
        r.copy->setEnabled(true);
    if (r.fail)
        r.fail->hide();
}

void DecodePage::showResult(const QString &code)
{
    if (!isValidCode(code))
    {
        showFailure(tr("Invalid code: must be exactly 6 digits, first digit non-zero."), "ERR_INVALID_CODE");
        return;
    }
    applyResult(m_cam, code);
    applyResult(m_up, code);
    m_log.code->setText(code); // overview keeps its permanent telemetry box
    m_log.copy->setEnabled(true);
    if (m_log.fail)
        m_log.fail->hide();
    m_camera->setPreviewCode(code);
}

void DecodePage::showFailure(const QString &reason, const QString &errCode)
{
    for (const ResultUi *r : {&m_cam, &m_up})
    {
        if (!r->code)
            continue;
        r->code->setText("------");
        r->code->setStyleSheet("color:#C2CCDF;");
        if (r->badge)
            setBadge(r->badge, "Could not read barcode", "red");
        if (r->copy)
            r->copy->setEnabled(false);
    }
    for (QFrame *f : {m_up.fail, m_log.fail})
    {
        if (!f)
            continue;
        if (auto *t = f->findChild<QLabel *>("failReason"))
            t->setText(reason.isEmpty() ? tr("Could not decode the barcode.") : reason);
        if (auto *c = f->findChild<QLabel *>("failCode"))
            c->setText(errCode.isEmpty() ? QString("ERR_UNKNOWN") : errCode);
        f->show();
    }
}

void DecodePage::resetResult()
{
    for (const ResultUi *r : {&m_cam, &m_up})
    {
        if (!r->code)
            continue;
        r->code->setText("------");
        r->code->setStyleSheet("color:#C2CCDF;");
        if (r->badge)
            setBadge(r->badge, "\xE2\x97\x8F  Waiting", "idle");
        if (r->copy)
            r->copy->setEnabled(false);
        if (r->fail)
            r->fail->hide();
    }
    if (m_log.fail)
        m_log.fail->hide();
}

void DecodePage::setCameraList(const QStringList &names)
{
    m_cameraBox->clear();
    m_cameraBox->addItems(names.isEmpty() ? QStringList{tr("No camera found")} : names);
}

void GeneratePage::setBarcodeImage(const QPixmap &pm) { m_view->setImage(pm); }

void GeneratePage::setCode(const QString &code, bool generate)
{
    if (code.isEmpty()) { m_input->clear(); return; }
    if (!generate) { m_input->setText(code); return; }   // prefill only; onTextChanged() shows any problem

    if (!isValidCode(code)) {
        m_input->clear();                                  // resets hint state first ...
        showInputError(m_input, m_hint, m_hintIcon,        // ... then shows the error
                       tr("Invalid code: must be exactly 6 digits, first digit not zero"));
        return;
    }
    m_input->setText(code);
    m_view->setCode(code);
    m_stageSeg->setCurrent(0);
    m_view->setAwaiting(false);
    setProperty("genCode", code);                          // the code that is actually shown in the preview
    m_save->setEnabled(true);
    m_copy->setEnabled(true);
}

void GeneratePage::onTextChanged(const QString &t)
{
    m_counter->setText(QString("%1 / 6").arg(t.size()));
    const bool valid = isValidCode(t);
    m_generate->setEnabled(valid);
    m_validChip->setVisible(valid);
    m_checkAction->setVisible(valid);

    setKind(m_input, "");
    m_hint->setText(tr("Must be exactly 6 digits, first digit not zero"));
    m_hint->setStyleSheet("font-size:13px; color:#434655;");
    m_hintIcon->setPixmap(iconPixmap("info", QColor("#0B4BCB"), 16));

    if (!t.isEmpty() && t.at(0) == QLatin1Char('0'))
        showInputError(m_input, m_hint, m_hintIcon, tr("First digit must not be zero"));
}

void GeneratePage::onGenerate()
{
    const QString code = m_input->text();
    if (!isValidCode(code)) {                              // defensive: button/Enter should already be blocked
        m_generate->setEnabled(false);
        showInputError(m_input, m_hint, m_hintIcon,
                       tr("Invalid code: must be exactly 6 digits, first digit not zero"));
        return;
    }

    m_view->setCode(code);
    m_stageSeg->setCurrent(0);
    m_view->setAwaiting(false);
    setProperty("genCode", code);
    m_save->setEnabled(true);
    m_copy->setEnabled(true);

    for (int i = m_sessions.size() - 1; i >= 0; --i)
        if (m_sessions[i].code == code) m_sessions.removeAt(i);
    m_sessions.prepend({code, "Code 128", QTime::currentTime().toString("HH:mm:ss")});
    while (m_sessions.size() > 3) m_sessions.removeLast();
    rebuildSessions();
    emit generateRequested(code);
}

void GeneratePage::onClear()
{
    m_input->clear();
    m_view->clear();
    setProperty("genCode", QString());
    m_save->setEnabled(false);
    m_copy->setEnabled(false);
    m_input->setFocus();
}

void GeneratePage::onSave()
{
    const QString code = property("genCode").toString();
    if (code.isEmpty()) {
        QMessageBox::warning(this, tr("Save barcode"), tr("Generate a barcode first."));
        return;
    }
    const QPixmap pm = m_view->renderBarcode();
    if (pm.isNull()) {
        QMessageBox::warning(this, tr("Save barcode"), tr("Could not render the barcode image."));
        return;
    }
    QString path = QFileDialog::getSaveFileName(this, tr("Save barcode"), code + ".png", tr("PNG image (*.png)"));
    if (path.isEmpty()) return;                            // user cancelled
    if (!path.endsWith(".png", Qt::CaseInsensitive)) path += ".png";
    if (!pm.save(path, "PNG"))
        QMessageBox::warning(this, tr("Save barcode"),
                             tr("Could not save the file:\n%1\n\nCheck the folder permissions and free disk space.").arg(path));
}

void GeneratePage::onCopy()
{
    if (m_copy->property("copied").toBool()) return;       // double-click guard
    const QPixmap pm = m_view->renderBarcode();
    if (pm.isNull()) {
        QMessageBox::warning(this, tr("Copy image"), tr("Could not render the barcode image."));
        return;
    }
    QApplication::clipboard()->setPixmap(pm);
    const QString old = m_copy->text();
    m_copy->setProperty("copied", true);
    m_copy->setText(tr(" Copied!"));
    QTimer::singleShot(1500, this, [this, old] {
        m_copy->setText(old);
        m_copy->setProperty("copied", false);
    });
}

void GeneratePage::rebuildSessions()
{
    while (m_sessionRow->count()) {
        QLayoutItem *it = m_sessionRow->takeAt(0);
        if (it->widget()) it->widget()->deleteLater();
        delete it;
    }
    if (m_sessions.isEmpty()) {
        m_sessionRow->addWidget(T(tr("No barcodes generated yet"), 12, 400, "#737686", true), 1);
        return;
    }
    for (int i = 0; i < m_sessions.size(); ++i) {
        const Session &s = m_sessions[i];
        auto *b = new QPushButton;
        b->setObjectName("tile");
        b->setProperty("active", i == 0);
        b->setCursor(Qt::PointingHandCursor);
        b->setMinimumHeight(70);
        auto *l = vbox(b, 14, 10, 14, 10, 4);
        auto *top = hrow();
        auto *code = T(s.code, 16, 600, i == 0 ? "#0B4BCB" : "#0B1C30", true);
        top->addWidget(code);
        top->addStretch();
        top->addWidget(chip(i == 0 ? "Active" : "Logged", i == 0 ? "blue" : "neutral"));
        l->addLayout(top);
        l->addWidget(T(s.symbology + " \xE2\x80\xA2 " + s.time, 11, 400, "#737686", true));
        for (auto *w : b->findChildren<QWidget *>()) w->setAttribute(Qt::WA_TransparentForMouseEvents);
        connect(b, &QPushButton::clicked, this, [this, c = s.code] { setCode(c, true); });
        m_sessionRow->addWidget(b, 1);
    }
}

// Pills and chips are painted by hand: Qt stylesheets drop border-radius on QLabel when
// the colour comes from a dynamic-property rule, so custom painting is the reliable way.
class Badge : public QLabel {
public:
    Badge(const QString &text, const QString &kind, bool isPill) : QLabel(text), m_pill(isPill)
    {
        setProperty("kind", kind);
        QFont f = isPill ? QFont() : monoFont(11, true);
        f.setPixelSize(isPill ? 12 : 11);
        f.setWeight(QFont::DemiBold);
        setFont(f);
        setContentsMargins(isPill ? 11 : 8, isPill ? 3 : 2, isPill ? 11 : 8, isPill ? 3 : 2);
        restyle();
    }
protected:
    bool event(QEvent *e) override
    {
        if (e->type() == QEvent::DynamicPropertyChange) restyle();
        return QLabel::event(e);
    }
    void paintEvent(QPaintEvent *e) override
    {
        {
            QPainter p(this);
            p.setRenderHint(QPainter::Antialiasing);
            p.setBrush(m_bg);
            p.setPen(m_border.isValid() ? QPen(m_border, 1) : QPen(Qt::NoPen));
            const qreal r = m_pill ? height() / 2.0 : 6;
            p.drawRoundedRect(QRectF(rect()).adjusted(0.5, 0.5, -0.5, -0.5), r, r);
        }
        QLabel::paintEvent(e);
    }
private:
    void restyle()
    {
        const QString k = property("kind").toString();
        m_border = QColor();
        if (m_pill) {
            if (k == "white")      { m_bg = "#FFFFFF"; m_fg = "#0B1C30"; m_border = QColor("#E4E8F1"); }
            else if (k == "green") { m_bg = "#E3F6EC"; m_fg = "#0B7A4B"; }
            else if (k == "red")   { m_bg = "#FFDAD6"; m_fg = "#93000A"; }
            else if (k == "idle")  { m_bg = "#EEF2FA"; m_fg = "#5A6075"; }
            else                   { m_bg = "#E1EAFF"; m_fg = "#1D4ED8"; }      // tint
        } else {
            if (k == "neutral")    { m_bg = "#EEF2FA"; m_fg = "#434655"; m_border = QColor("#E2E7F1"); }
            else if (k == "white") { m_bg = "#FFFFFF"; m_fg = "#434655"; }
            else if (k == "green") { m_bg = "#E3F6EC"; m_fg = "#0B7A4B"; }
            else                   { m_bg = "#E1EAFF"; m_fg = "#1F4FCF"; }      // blue
        }
        setStyleSheet(QString("color:%1; background:transparent; border:none;").arg(m_fg.name()));
        update();
    }
    bool m_pill;
    QColor m_bg, m_fg, m_border;
};

// ======================================================= mock bars
void drawMockBars(QPainter &p, const QRect &r, const QString &codeIn, const QColor &color)
{
    const QString code = codeIn.isEmpty() ? QString("000000") : codeIn;
    const bool dense = r.width() >= 140;              // small glyphs use fewer, thicker bars
    QVector<int> w = {2, 1, 1, 2, 3, 2};                                    // start
    for (int i = 0; i < code.size(); ++i) {
        int d = code[i].digitValue();
        if (d < 0) d = 0;
        w << (d % 3) + 1 << ((d * 7 + i) % 3) + 1 << ((d + i * 2) % 2) + 1
          << ((d * 5 + 3) % 3) + 1 << ((d + i) % 2) + 1 << ((d * 3 + 1) % 3) + 1
;
        if (dense)
            w << ((d * 2 + i) % 3) + 1 << ((d + i * 3) % 2) + 1 << ((d * 4 + 1) % 3) + 1
              << ((d + 2 * i) % 2) + 1 << ((d * 6 + i) % 3) + 1 << ((d + 1) % 2) + 1;
    }
    w << 2 << 3 << 3 << 1 << 1 << 1 << 2;                                   // stop
    int total = 0;
    for (int v : w) total += v;

    p.save();
    p.setRenderHint(QPainter::Antialiasing, false);
    p.setPen(Qt::NoPen);
    p.setBrush(color);
    const double unit = double(r.width()) / total;
    double x = r.left();
    for (int i = 0; i < w.size(); ++i) {
        const double bw = w[i] * unit;
        if (i % 2 == 0)
            p.drawRect(QRect(qRound(x), r.top(), qMax(1, qRound(x + bw) - qRound(x)), r.height()));
        x += bw;
    }
    p.restore();
}

}

void BarcodeView::paintBarcodeCard(QPainter &p, const QRect &r) const
{
    p.fillRect(r, Qt::white);
    if (!m_image.isNull()) {
        const QPixmap s = m_image.scaled(r.size() - QSize(40, 24), Qt::KeepAspectRatio, Qt::SmoothTransformation);
        p.drawPixmap(r.center().x() - s.width() / 2, r.center().y() - s.height() / 2, s);
        return;
    }
    const int mx = qRound(r.width() * 0.10);
    const int barH = m_digits ? r.height() - 90 : r.height() - 56;
    drawMockBars(p, QRect(r.left() + mx, r.top() + 28, r.width() - 2 * mx, barH), m_code, QColor("#0B1C30"));
    if (m_digits) {
        QFont f = monoFont(22, true);
        f.setLetterSpacing(QFont::AbsoluteSpacing, 0);
        p.setFont(f);
        p.setPen(QColor("#0B1C30"));
        p.drawText(QRect(r.left(), r.top() + 28 + barH + 8, r.width(), 36), Qt::AlignCenter, spaced(m_code));
    }
}

QPixmap BarcodeView::renderBarcode() const
{

    QPixmap pm(base.size() * 2);
    pm.fill(Qt::white);

void BarcodeView::paintEvent(QPaintEvent *)
{
    QPainter p(this);
    p.setRenderHint(QPainter::Antialiasing);

    // stage + dot grid
    QPainterPath stage;
    stage.addRoundedRect(QRectF(rect()), 10, 10);
    p.fillPath(stage, QColor("#EEF3FF"));
    p.setClipPath(stage);
    p.setPen(QPen(QColor("#C9D6EE"), 1.5, Qt::SolidLine, Qt::RoundCap));
    for (int y = 12; y < height(); y += 16)
        for (int x = 12; x < width(); x += 16) p.drawPoint(x, y);
    p.setClipping(false);

    const QColor muted("#737686");
    QFont mf = monoFont(11);
    p.setFont(mf);

    // ruler row (top)
    p.drawPixmap(14, 16, iconPixmap("ruler", muted, 14));
    p.setPen(muted);
    p.drawText(QRect(32, 12, 150, 22), Qt::AlignVCenter | Qt::AlignLeft, "600 px width");
    p.drawText(QRect(width() - 182, 12, 150, 22), Qt::AlignVCenter | Qt::AlignRight, "240 px height");
    p.drawPixmap(width() - 28, 16, iconPixmap("ruler", muted, 14));

    const QString sym = "Active Symbology: CODE 128";
    const int sw = QFontMetrics(mf).horizontalAdvance(sym) + 22;
    const QRect symR((width() - sw) / 2, 11, sw, 24);
    p.setPen(Qt::NoPen);
    p.setBrush(QColor("#E3EAF8"));
    p.drawRoundedRect(symR, 6, 6);
    p.setPen(QColor("#434655"));
    p.drawText(symR, Qt::AlignCenter, sym);

    // bottom row
    p.setBrush(QColor("#0B4BCB"));
    p.setPen(Qt::NoPen);
    p.drawEllipse(QPoint(20, height() - 22), 3, 3);
    p.setPen(muted);
    p.drawText(QRect(30, height() - 34, 220, 22), Qt::AlignVCenter | Qt::AlignLeft, "Optical Density: 99.4%");
    p.drawText(QRect(width() - 262, height() - 34, 246, 22), Qt::AlignVCenter | Qt::AlignRight, "Standard Bumper: Left/Right 12mm");

    // center card
    const int cw = qMin(width() - 80, 450), ch = 236;
    const QRect c((width() - cw) / 2, (height() - ch) / 2, cw, ch);

    if (!hasBarcode() || m_awaiting) {
        p.setPen(QPen(QColor("#9DB4E0"), 1.6, Qt::DashLine));
        p.setBrush(QColor(255, 255, 255, 170));
        p.drawRoundedRect(c, 10, 10);
        QFont f; f.setPixelSize(15); f.setWeight(QFont::DemiBold);
        p.setFont(f);
        p.setPen(QColor("#434655"));
        p.drawText(c.adjusted(0, 0, 0, -22), Qt::AlignCenter, tr("Your barcode will appear here"));
        f.setPixelSize(12); f.setWeight(QFont::Normal);
        p.setFont(f);
        p.setPen(QColor("#737686"));
        p.drawText(c.adjusted(0, 34, 0, 0), Qt::AlignCenter, tr("Enter a 6-digit code and press Generate"));
        return;
    }

    for (int i = 5; i >= 1; --i) {                                          // soft shadow

static void drawChip(QPainter &p, const QRect &r, const QString &text, const QColor &bg, const QColor &fg,
                     const QColor &dot = QColor(), const QString &icon = QString())
{
    p.setPen(Qt::NoPen);
    p.setBrush(bg);
    p.drawRoundedRect(r, 6, 6);
    int x = r.left() + 10;
    if (dot.isValid()) { p.setBrush(dot); p.drawEllipse(QPoint(x + 3, r.center().y()), 4, 4); x += 14; }
    if (!icon.isEmpty()) { p.drawPixmap(x, r.center().y() - 7, iconPixmap(icon, fg, 14)); x += 20; }

    p.drawText(QRect(x, r.top(), r.right() - x, r.height()), Qt::AlignVCenter | Qt::AlignLeft, text);
}

void CameraView::paintEvent(QPaintEvent *)
{
    QPainter p(this);
    p.setRenderHint(QPainter::Antialiasing);
    QPainterPath clip;
    clip.addRoundedRect(QRectF(rect()), 10, 10);

    QRadialGradient bg(width() / 2.0, height() / 2.0, qMax(width(), height()) * 0.7);
    bg.setColorAt(0, QColor("#10243F"));
    bg.setColorAt(1, QColor("#070F1C"));
    p.fillRect(rect(), bg);
    if (!m_frame.isNull())
        p.drawPixmap(rect(), m_frame.scaled(size(), Qt::KeepAspectRatioByExpanding, Qt::SmoothTransformation));

    p.setPen(QPen(QColor(255, 255, 255, 14), 1.5, Qt::SolidLine, Qt::RoundCap));
    for (int y = 14; y < height(); y += 18)
        for (int x = 14; x < width(); x += 18) p.drawPoint(x, y);

    // scan frame (corner brackets)
    const int fw = int(width() * 0.74), fh = int(height() * 0.64);
    const QRect f((width() - fw) / 2, (height() - fh) / 2, fw, fh);
    p.setBrush(Qt::NoBrush);
    p.setPen(QPen(QColor("#1B52D6"), 4, Qt::SolidLine, Qt::FlatCap, Qt::MiterJoin));
    const int L = 30;
    auto corner = [&](QPoint c, int dx, int dy) {
        p.drawLine(c, c + QPoint(dx * L, 0));
        p.drawLine(c, c + QPoint(0, dy * L));
    };
    corner(f.topLeft() + QPoint(2, 2), 1, 1);
    corner(f.topRight() + QPoint(-2, 2), -1, 1);
    corner(f.bottomLeft() + QPoint(2, -2), 1, -1);
    corner(f.bottomRight() + QPoint(-2, -2), -1, -1);

    // sample barcode card
    const int cw = int(width() * 0.62), ch = int(height() * 0.47);
    const QRect bc((width() - cw) / 2, (height() - ch) / 2, cw, ch);
    p.setPen(Qt::NoPen);
    p.setBrush(Qt::white);
    p.drawRoundedRect(bc, 10, 10);
    const int mx = int(bc.width() * 0.09);
    drawMockBars(p, QRect(bc.left() + mx, bc.top() + 22, bc.width() - 2 * mx, bc.height() - 70), m_code, QColor("#0B1C30"));
    QFont df = monoFont(15, true);
    p.setFont(df);
    p.setPen(QColor("#0B1C30"));
    p.drawText(QRect(bc.left(), bc.bottom() - 40, bc.width(), 30), Qt::AlignCenter, spaced(m_code));

    // scan line
    const qreal t = m_frozen ? 0.5 : (qSin(m_phase) + 1) / 2;
    const int y = f.top() + 14 + int(t * (f.height() - 28));
    QLinearGradient glow(0, y - 26, 0, y + 26);
    glow.setColorAt(0, QColor(34, 211, 238, 0));
    glow.setColorAt(0.5, QColor(34, 211, 238, 55));
    glow.setColorAt(1, QColor(34, 211, 238, 0));
    p.fillRect(QRect(f.left(), y - 26, f.width(), 52), glow);
    QLinearGradient ln(f.left(), 0, f.right(), 0);
    ln.setColorAt(0, QColor(34, 211, 238, 0));
    ln.setColorAt(0.5, QColor(103, 232, 249, 255));
    ln.setColorAt(1, QColor(34, 211, 238, 0));
    p.setPen(QPen(QBrush(ln), 2));
    p.drawLine(f.left(), y, f.right(), y);

    // HUD chips
    p.setFont(monoFont(11));
    drawChip(p, QRect(14, 14, 252, 26), "LIVE SENSOR: 1920x1080@60Hz", QColor(0, 0, 0, 110), QColor("#E2E8F0"),
             m_running ? QColor("#34D399") : QColor("#94A3B8"));

             QColor(), "target");
    drawChip(p, QRect(width() - 224, height() - 40, 210, 26), "AGC: +2.4dB | EXP: 1/120s", QColor(0, 0, 0, 120), QColor("#CBD5E1"));

    if (e->mimeData()->hasUrls()) { m_hover = true; update(); e->acceptProposedAction(); }

void DropZone::dropEvent(QDropEvent *e)
{
    m_hover = false; update();
    for (const QUrl &u : e->mimeData()->urls()) {
        const QString suf = QFileInfo(u.toLocalFile()).suffix().toLower();
        if (u.isLocalFile() && (suf == "png" || suf == "jpg" || suf == "jpeg" || suf == "bmp")) {
            emit fileChosen(u.toLocalFile());
            return;
        }
    }
}

# Increment 1 — Barcode Generator and Reader

---

## Overview

The first increment focuses on developing a working version of the C++ desktop application with barcode generation and reading capabilities.

The application must be able to generate a barcode from a six-digit number, display it on the screen, save it as an image, and read the generated barcode through either an uploaded image or a laptop camera.

---

## Requirements

### 1. Barcode Generation

The application must:

- Accept a six-digit numeric input from the user.
- Ensure that the first digit is non-zero.
- Generate a barcode corresponding to the provided number.
- Display the generated barcode on the application screen.
- Save the generated barcode to disk as an image.

### 2. Barcode Reading

The application must be able to read a barcode generated through the application.

**A. Image Upload**

- Allow the user to select and upload a barcode image.
- Read and decode the barcode from the selected image.
- Display the decoded six-digit number on the screen.

**B. Laptop Camera**

- Allow the user to scan a barcode using the laptop camera.
- Detect and decode the barcode from the camera feed.
- Display the decoded six-digit number on the screen.

---

## Technical Constraints

- Existing non-standard libraries must not be used for barcode generation.
- Libraries for image processing, file handling, and other supporting operations are permitted.
- The application must be developed in C++.
- All submissions must be made through GitHub.

# Contributing Guidelines

This document explains how team members can contribute to **Bar-Code-System-CPP**.

## 1. Clone the Repository

Clone the repository and navigate into the project directory:

```bash
git clone https://github.com/Syntrojex/Bar-Code-System-CPP.git
cd Bar-Code-System-CPP
```

## 2. Switch to Your Assigned Branch

Switch to your assigned branch. Replace `your-branch` with your actual branch name.

```bash
git checkout your-branch
```

## 3. Synchronize with Main

Before starting a new task, update your assigned branch with the latest changes from `main`.

```bash
git fetch origin
git merge origin/main
```

Resolve any merge conflicts before continuing.

## 4. Make Your Changes

Add or modify the files required for your assigned task.

Follow the existing project structure and coding conventions. Avoid modifying unrelated files.

## 5. Stage and Commit Changes

Check your changes:

```bash
git status
```

Stage your files:

```bash
git add .
```

Commit your changes with a meaningful message:

```bash
git commit -m "Add barcode generation"
```

## 6. Push Your Branch

Push your changes to your assigned branch:

```bash
git push origin your-branch
```

## 7. Create a Pull Request

Submit your changes for integration into `main`.

1. Open the project repository on GitHub.
2. Select **Compare & pull request**.
3. Set the base branch to `main`.
4. Select your assigned branch as the compare branch.
5. Add a meaningful title and description.
6. Create the Pull Request.

## 8. Review and Merge

The project lead will review and merge Pull Requests into `main`.

Team members should not directly push changes to `main`.

## 9. Start Your Next Task

After your Pull Request has been merged, synchronize your assigned branch with the latest `main` before starting another task.

Repeat the synchronization commands from Section 3, then continue with the same workflow.

## General Guidelines

- Always work on your assigned branch.
- Keep commits meaningful and focused.
- Follow the existing C++ coding conventions.
- Test your changes before submitting a Pull Request.
- Keep Pull Requests limited to the assigned task.
- Coordinate with the project lead when necessary.

Thank you for contributing to the project.

# Bar-Code-System-CPP

A C++ desktop application designed to simplify inventory management and sales operations for small retail stores, supermarkets, and corner shops.

---

## Project Overview

Bar-Code-System-CPP is an academic term project developed for the **Introduction to Software Engineering (ISE)** course at FAST-NUCES.

The application aims to provide a simple and efficient system through which store employees can manage inventory and record sales with minimal manual input.

Cashiers will be able to scan product barcodes, retrieve product information, add items to orders, and generate bills. Managers will also be able to generate barcodes for products in the store.

The project is being developed incrementally, with a focus on building a reliable, user-friendly, and maintainable desktop application.

---

## Key Features

- **Barcode Generation** — Generate barcodes for products.
- **Barcode Scanning** — Read barcodes using a laptop camera.
- **Image-Based Reading** — Upload barcode images for decoding.

---

## Technology Stack

- **C++** — Core application logic.
- **Qt** — Desktop graphical user interface.
- **CMake** — Build system and project configuration.
- **vcpkg** — C++ dependency management.
- **OpenCV** — Image processing and camera capture.

---

## Project Structure

```text
Bar-Code-System-CPP/
│
├── Frontend/             # Desktop user interface
├── Backend/              # Core application logic
├── Project-Planning/     # Increment-wise project documentation
│
├── CMakeLists.txt        # Build configuration
├── CMakePresets.json     # CMake build presets
├── vcpkg.json            # Dependency manifest
├── .gitignore
└── README.md
```

---

## Development Workflow

The project follows a collaborative GitHub workflow:

1. Each team member works on an assigned branch.
2. Changes are committed and pushed to GitHub.
3. A Pull Request is opened against the `main` branch.
4. Changes are reviewed before being merged into the main project.
