name: Sync All Branches

on:
  push:
  workflow_dispatch:

permissions:
  contents: write

concurrency:
  group: sync-all-branches
  cancel-in-progress: false

jobs:
  sync:
    name: Sync Branches with Default Branch
    runs-on: ubuntu-latest

    if: github.event_name == 'workflow_dispatch' || github.ref == format('refs/heads/{0}', github.event.repository.default_branch)

    steps:
      - name: Checkout Repository
        uses: actions/checkout@v4
        with:
          fetch-depth: 0
          token: ${{ secrets.GITHUB_TOKEN }}

      - name: Configure Git
        run: |
          git config user.name "github-actions[bot]"
          git config user.email "41898282+github-actions[bot]@users.noreply.github.com"

      - name: Detect and Sync All Branches
        shell: bash
        env:
          DEFAULT_BRANCH: ${{ github.event.repository.default_branch }}
        run: |
          set -u

          git fetch origin --prune

          echo "Default branch: $DEFAULT_BRANCH"

          branches=()
          failed=()

          while IFS= read -r branch; do
            [[ -z "$branch" ]] && continue
            [[ "$branch" == "HEAD" ]] && continue
            [[ "$branch" == "$DEFAULT_BRANCH" ]] && continue

            branches+=("$branch")
          done < <(git for-each-ref \
            --format='%(refname:strip=3)' \
            refs/remotes/origin)

          if (( ${#branches[@]} == 0 )); then
            echo "No branches found to synchronize."
            exit 0
          fi

          echo "Found ${#branches[@]} branches."

          for branch in "${branches[@]}"; do
            echo "::group::Syncing $branch"

            if ! git checkout -B "$branch" "origin/$branch"; then
              echo "::warning::Checkout failed for $branch."
              failed+=("$branch (checkout failed)")
              echo "::endgroup::"
              continue
            fi

            if git merge --no-edit "origin/$DEFAULT_BRANCH"; then
              if git push origin "HEAD:$branch"; then
                echo "Successfully synced $branch."
              else
                echo "::warning::Push failed for $branch."
                failed+=("$branch (push failed)")
              fi
            else
              echo "::warning::Merge conflict in $branch."
              git merge --abort || true
              failed+=("$branch (merge conflict)")
            fi

            echo "::endgroup::"
          done

          if (( ${#failed[@]} > 0 )); then
            echo "::error::Some branches could not be synchronized:"
            printf ' - %s\n' "${failed[@]}"
            exit 1
          fi

          echo "All branches successfully synchronized."

// Generates an EAN-8 barcode (the symbology Backend/Decoder reads).
// The 6-digit input is encoded as: "0" + <6 digits> + <mod-10 check digit>.
class BarcodeGenerator
{
public:
    // Validates the six-digit input (first digit non-zero) and renders a 600x240 EAN-8 image.
    // humanReadable = print the 8 digits under the bars.
    BarcodeResult generate(const QString &number, bool humanReadable = true) const;

    // The 8-digit EAN-8 payload for a valid 6-digit code ("" if the code is invalid).
    static QString ean8Payload(const QString &six);
};

﻿#include "barcode/BarcodeGenerator.h"
#include <QFont>
#include <QPainter>
#include <QRegularExpression>

namespace {
// EAN-8 left-hand ("L") digit patterns; right-hand digits use the bitwise complement.
const char *const kLeft[10] = {"0001101", "0011001", "0010011", "0111101", "0100011",
                               "0110001", "0101111", "0111011", "0110111", "0001011"};

bool validSix(const QString &s)
{
    static const QRegularExpression re("^[1-9][0-9]{5}$");
    return re.match(s).hasMatch();
}

QString modulesFor(const QString &eight)
{
    QString bits = "101";
    for (int i = 0; i < 4; ++i)
        bits += QLatin1String(kLeft[eight.at(i).digitValue()]);
    bits += "01010";
    for (int i = 4; i < 8; ++i) {
        QString p = QLatin1String(kLeft[eight.at(i).digitValue()]);
        for (QChar &c : p) c = (c == QLatin1Char('1')) ? QLatin1Char('0') : QLatin1Char('1');
        bits += p;
    }
    bits += "101";
    return bits;                                    // 3 + 28 + 5 + 28 + 3 = 67 modules
}
} // namespace

QString BarcodeGenerator::ean8Payload(const QString &six)
{
    if (!validSix(six)) return QString();
    const QString d = "0" + six;                    // 7 digits
    int sum = 0;
    for (int i = 0; i < 7; ++i)
        sum += d.at(i).digitValue() * ((i % 2 == 0) ? 3 : 1);
    return d + QChar('0' + (10 - sum % 10) % 10);
}

BarcodeResult BarcodeGenerator::generate(const QString &number, bool humanReadable) const
{
    BarcodeResult res;
    if (!validSix(number)) {
        res.success = false;
        res.error = "Input must be exactly 6 digits, with the first digit non-zero.";
        return res;
    }

    const QString eight = ean8Payload(number);
    const QString bits = modulesFor(eight);

    const int w = 600;
    const int h = 240;
    const int module = 7;                           // px per module -> 469 px of bars, ~9 modules of quiet zone each side
    const int x0 = (w - bits.size() * module) / 2;
    const int barTop = 26;
    const int barHeight = humanReadable ? 150 : 188;

    QImage img(w, h, QImage::Format_RGB32);
    img.fill(Qt::white);

    QPainter painter(&img);
    painter.setRenderHint(QPainter::Antialiasing, false);
    painter.setPen(Qt::NoPen);
    painter.setBrush(Qt::black);
    for (int i = 0; i < bits.size(); ++i) {
        if (bits.at(i) == QLatin1Char('1'))
            painter.drawRect(x0 + i * module, barTop, module, barHeight);
    }

    if (humanReadable) {
        painter.setRenderHint(QPainter::TextAntialiasing, true);
        QFont font("Consolas");
        font.setStyleHint(QFont::Monospace);
        font.setPixelSize(28);
        font.setBold(true);
        painter.setFont(font);
        painter.setPen(Qt::black);
        const int textTop = barTop + barHeight + 8;
        for (int i = 0; i < 8; ++i) {
            const int cell = (i < 4) ? (3 + 7 * i) : (36 + 7 * (i - 4));   // module index of this digit's 7-module cell
            painter.drawText(QRect(x0 + cell * module, textTop, 7 * module, 40), Qt::AlignCenter, QString(eight.at(i)));
        }
    }
    painter.end();

    res.success = true;
    res.image = img;
    return res;
}

#include <iostream>
#include <cmath>
#include <opencv2/imgproc.hpp>
#include "Decoder.h"
using namespace std;
const char *const Decoder::patterns[10] = {"0001101", "0011001", "0010011", "0111101", "0100011",
                                           "0110001", "0101111", "0111011", "0110111", "0001011"};
std::vector<double> Decoder::scanLineRuns(const cv::Mat &barcode) const
{
    if (barcode.empty() || barcode.type() != CV_8UC1 || barcode.rows < 5)
    {
        return {};
    }
    int y0 = barcode.rows * 4 / 10;
    int y1 = barcode.rows * 6 / 10;
    cv::Mat line;
    cv::reduce(barcode.rowRange(y0, y1), line, 0, cv::REDUCE_AVG, CV_32F);
    double low, high;
    cv::minMaxLoc(line, &low, &high);
    float T = (float)((low + high) / 2);
    vector<double> edges;
    for (int i = 0; i + 1 < line.cols; i++)
    {
        float a = line.at<float>(0, i);
        float b = line.at<float>(0, i + 1);
        if ((a < T) != (b < T))
        {
            edges.push_back(i + (T - a) / (b - a));
        }
    }
    if (edges.size() < 2)
        return {};
    vector<double> runs;
    for (size_t j = 0; j + 1 < edges.size(); j++)
    {
        runs.push_back(edges[j + 1] - edges[j]);
    }
    return runs;
}
std::string Decoder::runsToModules(const std::vector<double> &runs) const
{
    if (runs.size() != stripes)
        return "";
    double total = 0;
    for (double r : runs)
        total += r;
    double module = total / modules;
    string bits;
    for (size_t i = 0; i < runs.size(); i++)
    {
        int n = (int)lround(runs[i] / module);
        if (n < 1 || n > 4)
            return "";
        bits += string(n, i % 2 == 0 ? '1' : '0');
    }
    return (bits.size() == modules) ? bits : "";
}
int Decoder::digitMatch(const std::string &seven, bool right)
{
    for (int i = 0; i < 10; i++)
    {
        string pattern = patterns[i];
        if (right)
        {
            for (char &c : pattern)
                c = (c == '1') ? '0' : '1';
        }
        if (pattern == seven)
            return i;
    }
    return -1;
}
std::string Decoder::modulesToDigits(const std::string &bits) const
{
    if (bits.size() != modules || bits.substr(0, 3) != "101" || bits.substr(64, 3) != "101" || bits.substr(31, 5) != "01010")
        return "";
    string digits;
    // left slot of barcode
    for (int i = 0; i < 4; i++)
    {
        int d = digitMatch(bits.substr(3 + 7 * i, 7), false);
        if (d < 0)
            return "";
        digits += char('0' + d);
    }
    // right slot of barcode
    for (int j = 0; j < 4; j++)
    {
        int d = digitMatch(bits.substr(36 + 7 * j, 7), true);
        if (d < 0)
            return "";
        digits += char('0' + d);
    }
    return digits;
}
bool Decoder::isCheckDigit(const std::string &d) const
{
    int sum = 0;
    for (int i = 0; i < 7; i++)
        sum += (d[i] - '0') * ((i % 2 == 0) ? 3 : 1);
    return (10 - sum % 10) % 10 == d[7] - '0';
}
bool Decoder::decodeSix(const std::string &bits, std::string &six) const
{
    six.clear();
    badCheckDigit = false;
    for (int a = 0; a < 2; a++)
    {
        string b = (a == 0) ? bits : string(bits.rbegin(), bits.rend());
        string d = modulesToDigits(b);
        if (d.size() != 8)
            continue;
        if (!isCheckDigit(d))
        {
            badCheckDigit = true;
            continue;
        }
        if (d[0] == '0' && d[1] != '0')
        {
            six = d.substr(1, 6);
            badCheckDigit = false;
            return true;
        }
    }
    return false;
}
bool Decoder::decodeCropping(const cv::Mat &gray, std::string &six) const
{
    badCheckDigit = false;
    six.clear();
    if (gray.empty() || gray.type() != CV_8UC1) return false;
    bool badCheck = false;
    // Thin bands tolerate tilted bars and avoid text/scratches crossing one row.
    for (int percent : {50, 40, 60, 30, 70, 20, 80}) {
        const int y = gray.rows * percent / 100;
        const int top = std::max(0, y - 3);
        const int bottom = std::min(gray.rows, y + 4);
        cv::Mat band;
        cv::copyMakeBorder(gray.rowRange(top, bottom), band, 3, 3, 12, 12,
                           cv::BORDER_CONSTANT, cv::Scalar(255));
        const auto runs = scanLineRuns(band);
        // A barcode can be surrounded by other dark objects on the scan line.
        for (size_t start = 0; start + stripes <= runs.size(); start += 2) {
            const vector<double> candidate(runs.begin() + start, runs.begin() + start + stripes);
            const string bits = runsToModules(candidate);
            if (!bits.empty() && decodeSix(bits, six)) return true;
            badCheck = badCheck || badCheckDigit;
        }
    }
    badCheckDigit = badCheck;
    return false;
}
bool Decoder::isBadCheck() const
{
    return badCheckDigit;
}

bool CameraScanner::openCamera(int cameraIndex, QString *error, QSize resolution)
{
#ifdef HAVE_OPENCV
    closeCamera();
    if (error) error->clear();

        if (error) *error = "No camera selected.";
        return false;
    }

#ifdef _WIN32
    const bool opened = capture.open(cameraIndex, cv::CAP_DSHOW);   // DirectShow: fast start, same backend as Backend/Scanner

    const bool opened = capture.open(cameraIndex);

    if (opened) {

namespace {
// QImage -> BGR cv::Mat (transparent pixels are flattened onto white first).
cv::Mat toBgr(const QImage &src)
{
    QImage flat = src;
    if (src.hasAlphaChannel()) {
        flat = QImage(src.size(), QImage::Format_RGB32);
        flat.fill(Qt::white);
        QPainter p(&flat);
        p.drawImage(0, 0, src);
    }
    const QImage rgb = flat.convertToFormat(QImage::Format_RGB888);
    const cv::Mat view(rgb.height(), rgb.width(), CV_8UC3, const_cast<uchar *>(rgb.constBits()),
                       static_cast<size_t>(rgb.bytesPerLine()));
    cv::Mat bgr;
    cv::cvtColor(view, bgr, cv::COLOR_RGB2BGR);
    return bgr;
}
} // namespace
#endif

ImageScanResult ImageScanner::scan(const QImage &image) const
{
    ImageScanResult res;
    if (image.isNull()) {
        res.success = false;
        res.error = "Invalid or empty image provided.";
        return res;
    }

#ifndef HAVE_OPENCV
    res.error = "This build has no OpenCV, so barcode images cannot be decoded. Install OpenCV and reconfigure CMake.";
    return res;
#else
    try {
    QElapsedTimer timer;
    timer.start();

    const cv::Mat original = toBgr(image);
    Scanner scanner;                                // only its frame decoder is used (no camera is opened)
    Decoder decoder;

    // try the image as it is, then scaled to a size the barcode finder handles well
    std::vector<double> scales{1.0};
    const int longest = std::max(original.cols, original.rows);
    if (longest > 1400) scales.push_back(1000.0 / longest);
    if (longest < 500) scales.push_back(2.0);

    bool badCheck = false;
    for (double s : scales) {
        cv::Mat bgr;
        if (s == 1.0) bgr = original;
        else cv::resize(original, bgr, cv::Size(), s, s, s < 1.0 ? cv::INTER_AREA : cv::INTER_CUBIC);

        std::string six;
        cv::Rect box;
        if (scanner.frameRead(six, box, bgr)) {
            res.success = true;
            res.text = QString::fromStdString(six);
            if (box.area() > 0)
                res.box = QRect(qRound(box.x / s), qRound(box.y / s), qRound(box.width / s), qRound(box.height / s));
            res.latencyMs = static_cast<int>(timer.elapsed());
            return res;
        }

        // the image may be just the barcode itself (clean export): decode the whole picture
        cv::Mat gray, padded;
        cv::cvtColor(bgr, gray, cv::COLOR_BGR2GRAY);
        cv::copyMakeBorder(gray, padded, 40, 40, 60, 60, cv::BORDER_CONSTANT, cv::Scalar(255));
        if (decoder.decodeCropping(padded, six)) {
            res.success = true;
            res.text = QString::fromStdString(six);
            res.box = QRect(0, 0, image.width(), image.height());
            res.latencyMs = static_cast<int>(timer.elapsed());
            return res;
        }
        badCheck = badCheck || decoder.isBadCheck();
    }

    res.latencyMs = static_cast<int>(timer.elapsed());
    res.error = badCheck ? "Check digit mismatch: the barcode is damaged or is not a valid EAN-8."
                         : "No readable EAN-8 barcode found in this image.";
    return res;
    } catch (const cv::Exception &e) {
        res.error = QString("Image processing failed: %1").arg(QString::fromStdString(e.what()));
        return res;
    }
#endif
}

#include "Scanner.h"
#include <opencv2/imgproc.hpp>
using namespace std;
Scanner::Scanner(int idx, int width, int height)
{
    index = idx;
    this->width = width;
    this->height = height;
}
bool Scanner::openCamera()
{
#ifdef _WIN32
    camera.open(index, cv::CAP_DSHOW);
#else
    camera.open(index);
#endif
    if (!camera.isOpened())
    {
        errorMsg = "Camera cannot open";
        return false;
    }
    camera.set(cv::CAP_PROP_FOURCC, cv::VideoWriter::fourcc('M', 'J', 'P', 'G'));
    camera.set(cv::CAP_PROP_FRAME_WIDTH, width);
    camera.set(cv::CAP_PROP_FRAME_HEIGHT, height);
    camera.set(cv::CAP_PROP_FPS, 30);
    return true;
}
bool Scanner::find(const cv::Mat &gray, cv::Rect &box) const
{
    cv::Mat gx, gy, score, mask;
    cv::Sobel(gray, gx, CV_32F, 1, 0, 3);
    cv::Sobel(gray, gy, CV_32F, 0, 1, 3);
    score = cv::max(cv::abs(gx) - cv::abs(gy), 0.0);
    score.convertTo(score, CV_8U, 0.25);
    cv::blur(score, score, cv::Size(9, 9));
    cv::threshold(score, mask, 0, 255, cv::THRESH_BINARY | cv::THRESH_OTSU);
    cv::Mat k = cv::getStructuringElement(cv::MORPH_RECT, cv::Size(21, 7));
    cv::morphologyEx(mask, mask, cv::MORPH_CLOSE, k);
    cv::erode(mask, mask, cv::Mat(), cv::Point(-1, -1), 4);
    cv::dilate(mask, mask, cv::Mat(), cv::Point(-1, -1), 4);
    vector<vector<cv::Point>> contours;
    cv::findContours(mask, contours, cv::RETR_EXTERNAL, cv::CHAIN_APPROX_SIMPLE);
    cv::Rect Best;
    double bestArea = 0;
    for (const vector<cv::Point> &c : contours)
    {
        double a = cv::contourArea(c);
        if (a > bestArea)
        {
            bestArea = a;
            Best = cv::boundingRect(c);
        }
    }
    if (bestArea > 2000 && Best.width > Best.height && Best.height > 40)
    {
        box = Best;
        return true;
    }
    return false;
}
bool Scanner::frameRead(std::string &six, cv::Rect &box, const cv::Mat &frame)
{
    box = cv::Rect();
    six.clear();
    errorMsg.clear();
    if (frame.empty())
    {
        errorMsg = "Camera stopped sending frames";
        return false;
    }
    cv::Mat gray;
    if (frame.channels() == 1) gray = frame;
    else if (frame.channels() == 3) cv::cvtColor(frame, gray, cv::COLOR_BGR2GRAY);
    else if (frame.channels() == 4) cv::cvtColor(frame, gray, cv::COLOR_BGRA2GRAY);
    else return false;
    // Decode across several scan lines before relying on the contour detector.
    if (decoder.decodeCropping(gray, six)) {
        cv::Rect detected;
        box = find(gray, detected) ? detected : cv::Rect(0, 0, gray.cols, gray.rows);
        return true;
    }
    cv::Rect Best;
    if (!find(gray, Best)) {
        cv::Mat rotated;
        cv::rotate(gray, rotated, cv::ROTATE_90_CLOCKWISE);
        if (decoder.decodeCropping(rotated, six)) {
            box = cv::Rect(0, 0, gray.cols, gray.rows);
            return true;
        }
        return false;
    }
    box = Best;
    cv::Rect pad(Best.x - 15, Best.y, Best.width + 30, Best.height);
    pad &= cv::Rect(0, 0, gray.cols, gray.rows);
    cv::Mat barcode = gray(pad).clone();
    return decoder.decodeCropping(barcode, six);
}
bool Scanner::scan(std::string &sixDigits)
{
    sixDigits.clear();
    if (!camera.isOpened() && !openCamera()) return false;
    const auto start = cv::getTickCount();
    while ((cv::getTickCount() - start) / cv::getTickFrequency() < 30.0) {
        cv::Mat frame;
        if (!camera.read(frame) || frame.empty()) {
            errorMsg = "Camera stopped sending frames";
            camera.release();
            return false;
        }
        cv::Rect box;
        if (frameRead(sixDigits, box, frame)) {
            camera.release();
            return true;
        }
    }
    camera.release();
    errorMsg = "No barcode found within 30 seconds";
    return false;
}
std::string Scanner::errMessage() const
{
    return errorMsg;
}
Scanner::~Scanner()
{
    camera.release();
}

{
    QObject::connect(b, &QPushButton::clicked, b, [b, code] {
        if (b->property("copied").toBool()) return;     // double-click guard
        QApplication::clipboard()->setText(code->text());
        const QString old = b->text();
        b->setProperty("copied", true);
        b->setText(" Copied!");
        QTimer::singleShot(1300, b, [b, old] {
            b->setText(old);
            b->setProperty("copied", false);
        });
    });
}

static QFrame *resultHeader(const QString &title, QLabel *badge, int px = 18)
{
    auto *f = new QFrame;

    h->addWidget(T(title, px, 600, "#0B1C30"));
    h->addStretch();
    h->addWidget(badge);
    return f;
}

static QPixmap miniBarPixmap(const QString &code, QSize s)
{
    QPixmap pm(s);
    pm.fill(Qt::white);
    QPainter p(&pm);
    drawMockBars(p, QRect(8, 8, s.width() - 16, s.height() - 16), code, QColor("#0B1C30"));
    return pm;
}

void DecodePage::loadImage(const QString &path)
{
    const QFileInfo fi(path);
    if (!fi.exists() || !fi.isFile()) {
        m_path.clear(); m_loaded->hide();
        showFailure(tr("File not found."), "ERR_FILE_NOT_FOUND");
        return;
    }
    if (fi.size() > 32LL * 1024 * 1024) {
        m_path.clear(); m_loaded->hide();
        showFailure(tr("File exceeds the 32 MB limit."), "ERR_FILE_TOO_LARGE");
        return;
    }
    QPixmap pm(path);
    if (pm.isNull()) {
        m_path.clear(); m_loaded->hide();
        showFailure(tr("Unsupported or corrupted image."), "ERR_BAD_IMAGE");
        return;
    }
    m_path = path;
    m_thumb->setPixmap(pm.scaled(QSize(54, 44), Qt::KeepAspectRatio, Qt::SmoothTransformation));
    m_fileName->setText(fi.fileName());
    m_fileMeta->setText(QString("%1 KB &nbsp;\xE2\x80\xA2&nbsp; <span style='color:#0B4BCB'>%2 \xC3\x97 %3 px</span>")
                            .arg(qMax<qint64>(1, fi.size() / 1024)).arg(pm.width()).arg(pm.height()));
    m_loaded->show();
    resetResult();
}

// ================================================================== result state (all screens)
static const QString kDash = QString::fromUtf8("\xE2\x80\x94");

void DecodePage::applyResult(const ResultUi &r, const QString &code)
{
    if (!r.code) return;
    r.code->setText(code);
    r.code->setStyleSheet("color:#0B1C30;");
    if (r.badge) setBadge(r.badge, r.detectedWording ? "\xE2\x97\x8F Barcode Detected" : "\xE2\x97\x8F  Success", r.okKind);
    if (r.copy) r.copy->setEnabled(true);
    if (r.fail) r.fail->hide();
}

void DecodePage::clearDetails(const ResultUi &r)
{
    for (QLabel *l : {r.bounds, r.stamp, r.latency, r.payload})
        if (l) l->setText(kDash);
}

void DecodePage::showResult(const QString &code, const ScanInfo &info)
{
    if (!isValidCode(code)) {
        showFailure(tr("Invalid code: must be exactly 6 digits, first digit non-zero."), "ERR_INVALID_CODE");
        return;
    }
    applyResult(m_cam, code);
    applyResult(m_up, code);
    applyResult(m_log, code);
    m_log.copy->setEnabled(true);
    if (m_log.fail) m_log.fail->hide();
    for (QWidget *w : m_okOnly) w->show();

    const QString stamp = QTime::currentTime().toString("HH:mm:ss.zzz");
    const QString latency = info.latencyMs >= 0 ? QString("%1 ms").arg(info.latencyMs) : kDash;
    const QString box = info.box.isEmpty() ? kDash : QString("%1 \xC3\x97 %2 px").arg(info.box.width()).arg(info.box.height());
    const QString payload = info.payload.isEmpty() ? kDash : info.payload;
    if (m_cam.bounds) m_cam.bounds->setText(box);
    if (m_cam.stamp) m_cam.stamp->setText(stamp);
    if (m_cam.latency) m_cam.latency->setText(latency);
    if (m_up.bounds) m_up.bounds->setText(box);
    if (m_up.latency) m_up.latency->setText(latency);
    if (m_up.payload) m_up.payload->setText(payload);
    if (m_log.latency) m_log.latency->setText(latency);
    if (m_log.stamp) m_log.stamp->setText(stamp);
    if (m_autoCopy) QApplication::clipboard()->setText(code);
    m_history->insertRow(0);
    const QStringList values{QDateTime::currentDateTime().toString(Qt::ISODateWithMs), "EAN-8", code, payload, latency};
    for (int column = 0; column < values.size(); ++column)
        m_history->setItem(0, column, new QTableWidgetItem(values[column]));
    if (m_history->rowCount() > 1000) m_history->removeRow(1000);
}

void DecodePage::showFailure(const QString &reason, const QString &errCode)
{
    for (const ResultUi *r : {&m_cam, &m_up, &m_log}) {
        if (!r->code) continue;
        r->code->setText("------");
        r->code->setStyleSheet("color:#C2CCDF;");
        if (r->badge) setBadge(r->badge, "Could not read barcode", "red");
        if (r->copy) r->copy->setEnabled(false);
        clearDetails(*r);
    }
    for (QWidget *w : m_okOnly) w->hide();
    for (QFrame *f : {m_up.fail, m_log.fail}) {
        if (!f) continue;
        if (auto *t = f->findChild<QLabel *>("failReason"))
            t->setText(reason.isEmpty() ? tr("Could not decode the barcode.") : reason);
        if (auto *c = f->findChild<QLabel *>("failCode"))
            c->setText(errCode.isEmpty() ? QString("ERR_UNKNOWN") : errCode);
        f->show();
    }
}

void DecodePage::resetResult()
{
    for (const ResultUi *r : {&m_cam, &m_up, &m_log}) {
        if (!r->code) continue;
        r->code->setText("------");
        r->code->setStyleSheet("color:#C2CCDF;");
        if (r->badge) setBadge(r->badge, "\xE2\x97\x8F  Waiting", "idle");
        if (r->copy) r->copy->setEnabled(false);
        if (r->fail) r->fail->hide();
        clearDetails(*r);
    }
    for (QWidget *w : m_okOnly) w->hide();
    if (m_log.fail) m_log.fail->hide();
    if (m_log.code) m_log.code->setText("------");
    if (m_log.copy) m_log.copy->setEnabled(false);
    if (m_log.latency) m_log.latency->setText(kDash);
    if (m_log.stamp) m_log.stamp->setText(kDash);
}

void DecodePage::exportHistory()
{
    if (!m_history->rowCount()) {
        QMessageBox::information(this, tr("Export scans"), tr("No successful scans to export yet."));
        return;
    }
    QString path = QFileDialog::getSaveFileName(this, tr("Export scans"), "scans.csv", tr("CSV (*.csv)"));
    if (path.isEmpty()) return;
    if (!path.endsWith(".csv", Qt::CaseInsensitive)) path += ".csv";
    QSaveFile file(path);
    if (!file.open(QIODevice::WriteOnly | QIODevice::Text)) {
        QMessageBox::warning(this, tr("Export scans"), file.errorString());
        return;
    }
    QTextStream out(&file);
    out << "Timestamp,Symbology,Code,Payload,Latency\n";
    for (int row = 0; row < m_history->rowCount(); ++row) {
        for (int column = 0; column < m_history->columnCount(); ++column) {
            QString value = m_history->item(row, column)->text();
            value.replace('"', "\"\"");
            if (column) out << ',';
            out << '"' << value << '"';
        }
        out << '\n';
    }
    out.flush();
    if (out.status() != QTextStream::Ok || !file.commit())
        QMessageBox::warning(this, tr("Export scans"), tr("Could not write the scan log."));
}

    });

    connect(m_copy, &QPushButton::clicked, this, &GeneratePage::onCopy);
    connect(m_ocr, &ToggleSwitch::toggled, m_view, &BarcodeView::setShowDigits);
    connect(m_ocr, &ToggleSwitch::toggled, this, [this] {

void GeneratePage::setCode(const QString &code, bool generate)
{
    if (code.isEmpty()) { m_input->clear(); return; }
    if (!generate) { m_input->setText(code); return; }   // prefill only; onTextChanged() shows any problem

    if (!isValidCode(code)) {
        m_input->clear();                                  // resets hint state first ...
        showInputError(m_input, m_hint, m_hintIcon,        // ... then shows the error
                       tr("Invalid code: must be exactly 6 digits, first digit not zero"));
        return;
    }
    m_input->setText(code);
    onGenerate();                                          // same path as pressing the button
}

void GeneratePage::onTextChanged(const QString &t)
{
    m_counter->setText(QString("%1 / 6").arg(t.size()));
    const bool valid = isValidCode(t);
    m_generate->setEnabled(valid);
    m_validChip->setVisible(valid);
    m_checkAction->setVisible(valid);

    setKind(m_input, "");
    m_hint->setText(tr("Must be exactly 6 digits, first digit not zero"));
    m_hint->setStyleSheet("font-size:13px; color:#434655;");
    m_hintIcon->setPixmap(iconPixmap("info", QColor("#0B4BCB"), 16));

    if (!t.isEmpty() && t.at(0) == QLatin1Char('0'))
        showInputError(m_input, m_hint, m_hintIcon, tr("First digit must not be zero"));
}

void GeneratePage::onGenerate()
{
    const QString code = m_input->text();
    if (!isValidCode(code)) {                              // defensive: button/Enter should already be blocked
        m_generate->setEnabled(false);
        showInputError(m_input, m_hint, m_hintIcon,
                       tr("Invalid code: must be exactly 6 digits, first digit not zero"));
        return;
    }

    m_view->setCode(code);
    m_stageSeg->setCurrent(0);
    m_view->setAwaiting(false);
    setProperty("genCode", code);
    m_save->setEnabled(true);
    m_copy->setEnabled(true);

    for (int i = m_sessions.size() - 1; i >= 0; --i)
        if (m_sessions[i].code == code) m_sessions.removeAt(i);
    m_sessions.prepend({code, "EAN-8", QTime::currentTime().toString("HH:mm:ss")});
    while (m_sessions.size() > 3) m_sessions.removeLast();
    rebuildSessions();
    emit generateRequested(code);
}

void GeneratePage::onClear()
{
    m_input->clear();
    m_view->clear();
    setProperty("genCode", QString());
    m_save->setEnabled(false);
    m_copy->setEnabled(false);
    m_input->setFocus();
}

void GeneratePage::onSave()
{
    const QString code = property("genCode").toString();
    if (code.isEmpty()) {
        QMessageBox::warning(this, tr("Save barcode"), tr("Generate a barcode first."));
        return;
    }
    const QPixmap pm = m_view->renderBarcode();
    if (pm.isNull()) {
        QMessageBox::warning(this, tr("Save barcode"), tr("Could not render the barcode image."));
        return;
    }
    QString path = QFileDialog::getSaveFileName(this, tr("Save barcode"), code + ".png", tr("PNG image (*.png)"));
    if (path.isEmpty()) return;                            // user cancelled
    if (!path.endsWith(".png", Qt::CaseInsensitive)) path += ".png";
    if (!pm.save(path, "PNG"))
        QMessageBox::warning(this, tr("Save barcode"),
                             tr("Could not save the file:\n%1\n\nCheck the folder permissions and free disk space.").arg(path));
}

void GeneratePage::onCopy()
{
    if (m_copy->property("copied").toBool()) return;       // double-click guard
    const QPixmap pm = m_view->renderBarcode();
    if (pm.isNull()) {
        QMessageBox::warning(this, tr("Copy image"), tr("Could not render the barcode image."));
        return;
    }
    QApplication::clipboard()->setPixmap(pm);
    const QString old = m_copy->text();
    m_copy->setProperty("copied", true);
    m_copy->setText(tr(" Copied!"));
    QTimer::singleShot(1500, this, [this, old] {
        m_copy->setText(old);
        m_copy->setProperty("copied", false);
    });
}

void GeneratePage::rebuildSessions()
{
    while (m_sessionRow->count()) {
        QLayoutItem *it = m_sessionRow->takeAt(0);
        if (it->widget()) it->widget()->deleteLater();
        delete it;
    }
    if (m_sessions.isEmpty()) {
        m_sessionRow->addWidget(T(tr("No barcodes generated yet"), 12, 400, "#737686", true), 1);
        return;
    }
    for (int i = 0; i < m_sessions.size(); ++i) {
        const Session &s = m_sessions[i];
        auto *b = new QPushButton;
        b->setObjectName("tile");
        b->setProperty("active", i == 0);
        b->setCursor(Qt::PointingHandCursor);
        b->setMinimumHeight(70);
        auto *l = vbox(b, 14, 10, 14, 10, 4);
        auto *top = hrow();
        auto *code = T(s.code, 16, 600, i == 0 ? "#0B4BCB" : "#0B1C30", true);
        top->addWidget(code);
        top->addStretch();
        top->addWidget(chip(i == 0 ? "Active" : "Logged", i == 0 ? "blue" : "neutral"));
        l->addLayout(top);
        l->addWidget(T(s.symbology + " \xE2\x80\xA2 " + s.time, 11, 400, "#737686", true));
        for (auto *w : b->findChildren<QWidget *>()) w->setAttribute(Qt::WA_TransparentForMouseEvents);
        connect(b, &QPushButton::clicked, this, [this, c = s.code] { setCode(c, true); });
        m_sessionRow->addWidget(b, 1);
    }
}

    QMetaObject::invokeMethod(m_cameraWorker, [this] { delete m_scanner; m_scanner = nullptr; }, Qt::BlockingQueuedConnection);
    m_cameraThread->quit();
    m_cameraThread->wait();
}

void MainWindow::startCamera(int index)
{
#ifndef HAVE_OPENCV
    Q_UNUSED(index);
    m_decode->setCameraActive(false);
    m_decode->setCameraMessage(tr("This build has no OpenCV, so the webcam cannot be used. "
                                  "Install OpenCV and reconfigure CMake."), true);
    return;
#else
    const auto generation = ++m_cameraGeneration;
    const QSize resolution = m_decode->cameraResolution();
    m_failedReads = 0;
    m_missFrames = 0;
    m_lastCode.clear();
    m_readPending = false;
    QMetaObject::invokeMethod(m_cameraWorker, [this, index, resolution, generation] {
        QString error;
        const bool opened = m_scanner->openCamera(index, &error, resolution);
        QMetaObject::invokeMethod(this, [this, generation, opened, error] {
            if (generation != m_cameraGeneration) return;
            m_decode->setCameraActive(opened);
            if (opened) m_camTimer->start();
            else m_decode->setCameraMessage(error, true);
        }, Qt::QueuedConnection);
    }, Qt::QueuedConnection);
#endif
}

void MainWindow::stopCamera()
{
    if (m_camTimer) m_camTimer->stop();
    ++m_cameraGeneration;
    m_readPending = false;
    m_decode->setCameraActive(false);
    if (m_cameraWorker) QMetaObject::invokeMethod(m_cameraWorker, [this] { m_scanner->closeCamera(); }, Qt::QueuedConnection);
}

void MainWindow::pumpCamera()
{
    if (m_readPending) return;
    m_readPending = true;
    const auto generation = m_cameraGeneration;
    const bool decode = !m_decode->cameraView()->frozen();
    QMetaObject::invokeMethod(m_cameraWorker, [this, generation, decode] {
        const CameraFrameResult result = m_scanner->readFrame(decode);
        QMetaObject::invokeMethod(this, [this, generation, result] {
            if (generation != m_cameraGeneration) return;
            m_readPending = false;
            acceptCameraFrame(result);
        }, Qt::QueuedConnection);
    }, Qt::QueuedConnection);
}

void MainWindow::refreshCameras()
{
    if (m_decode->cameraActive()) return;
    QMetaObject::invokeMethod(m_cameraWorker, [this] {
        const QStringList names = CameraScanner::availableCameras();
        QMetaObject::invokeMethod(this, [this, names] {
            if (!m_decode->cameraActive()) m_decode->setCameraList(names);
        }, Qt::QueuedConnection);
    }, Qt::QueuedConnection);
}

void MainWindow::acceptCameraFrame(const CameraFrameResult &r)
{
    if (r.success && !r.frame.isNull()) {
        m_failedReads = 0;
        if (m_decode->cameraView()->frozen()) return;
        // mirrored for display only (like a mirror / selfie view); the original frame stays unflipped for decoding
        CameraView *view = m_decode->cameraView();
        const bool mirror = m_decode->mirrorCamera();
        view->setFrame(QPixmap::fromImage(mirror ? r.frame.mirrored(true, false) : r.frame));

        if (!r.decodedText.isEmpty()) {              // Backend/Scanner found and decoded an EAN-8 barcode in this frame
            m_missFrames = 0;
            const QRect b = r.box;
            view->setDetection(mirror ? QRect(r.frame.width() - (b.x() + b.width()), b.y(), b.width(), b.height()) : b, r.decodedText);
            if (r.decodedText != m_lastCode) {
                m_lastCode = r.decodedText;
                DecodePage::ScanInfo info;
                info.latencyMs = r.decodeMs;
                info.box = b.size();
                info.payload = BarcodeGenerator::ean8Payload(r.decodedText);
                m_decode->showResult(r.decodedText, info);
                m_decode->setCameraMessage(tr("Barcode detected: %1").arg(r.decodedText));
            }
        } else if (++m_missFrames > 8) {
            view->clearDetection();                  // barcode left the picture
            m_lastCode.clear();
        }
        return;
    }
    if (++m_failedReads >= 30) {                     // ~1 s without a frame: camera unplugged / busy
        stopCamera();
        m_decode->setCameraActive(false);
        m_decode->setCameraMessage(r.error.isEmpty() ? tr("Camera stopped delivering frames.") : r.error, true);
    }
}

void MainWindow::decodeFile(const QString &path)
{
    QImageReader reader(path);
    reader.setAutoTransform(true);                   // respect EXIF rotation of phone photos
    const QImage img = reader.read();
    if (img.isNull()) {
        m_decode->showFailure(tr("Could not read the image file: %1").arg(reader.errorString()), "ERR_BAD_IMAGE");
        return;
    }
    const ImageScanResult r = ImageScanner().scan(img);
    if (!r.success) {
        const QString code = r.error.contains("OpenCV") ? "ERR_NO_OPENCV"
                           : r.error.contains("Check digit") ? "ERR_CHECK_DIGIT" : "ERR_NO_BARCODE";
        m_decode->showFailure(r.error, code);
        return;
    }
    DecodePage::ScanInfo info;
    info.latencyMs = r.latencyMs;
    info.box = r.box.size();
    info.payload = BarcodeGenerator::ean8Payload(r.text);
    m_decode->showResult(r.text, info);
}

void MainWindow::makeSample(const QString &code, bool damaged)
{
    BarcodeResult r = BarcodeGenerator().generate(code, true);
    if (!r.success) {
        m_decode->showFailure(r.error, "ERR_SAMPLE");
        return;
    }
    if (damaged) {                                   // scratch the bars so the check/structure no longer matches
        QPainter p(&r.image);
        p.fillRect(250, 20, 30, 170, Qt::white);
        p.fillRect(380, 20, 12, 170, Qt::black);
    }
    if (!m_tmp) m_tmp = std::make_unique<QTemporaryDir>();
    const QString path = m_tmp->filePath(QString("sample_%1%2.png").arg(code, damaged ? "_damaged" : ""));
    if (!m_tmp->isValid() || !r.image.save(path, "PNG")) {
        m_decode->showFailure(tr("Could not write the sample image."), "ERR_SAMPLE");
        return;
    }
    m_decode->loadImage(path);                       // shows file name / size, like a real upload
    decodeFile(path);
}

// Pills and chips are painted by hand: Qt stylesheets drop border-radius on QLabel when
// the colour comes from a dynamic-property rule, so custom painting is the reliable way.
class Badge : public QLabel {
public:
    Badge(const QString &text, const QString &kind, bool isPill) : QLabel(text), m_pill(isPill)
    {
        setProperty("kind", kind);
        QFont f = isPill ? QFont() : monoFont(11, true);
        f.setPixelSize(isPill ? 12 : 11);
        f.setWeight(QFont::DemiBold);
        setFont(f);
        setContentsMargins(isPill ? 11 : 8, isPill ? 3 : 2, isPill ? 11 : 8, isPill ? 3 : 2);
        restyle();
    }
protected:
    bool event(QEvent *e) override
    {
        if (e->type() == QEvent::DynamicPropertyChange) restyle();
        return QLabel::event(e);
    }
    void paintEvent(QPaintEvent *e) override
    {
        {
            QPainter p(this);
            p.setRenderHint(QPainter::Antialiasing);
            p.setBrush(m_bg);
            p.setPen(m_border.isValid() ? QPen(m_border, 1) : QPen(Qt::NoPen));
            const qreal r = m_pill ? height() / 2.0 : 6;
            p.drawRoundedRect(QRectF(rect()).adjusted(0.5, 0.5, -0.5, -0.5), r, r);
        }
        QLabel::paintEvent(e);
    }
private:
    void restyle()
    {
        const QString k = property("kind").toString();
        m_border = QColor();
        if (m_pill) {
            if (k == "white")      { m_bg = "#FFFFFF"; m_fg = "#0B1C30"; m_border = QColor("#E4E8F1"); }
            else if (k == "green") { m_bg = "#E3F6EC"; m_fg = "#0B7A4B"; }
            else if (k == "red")   { m_bg = "#FFDAD6"; m_fg = "#93000A"; }
            else if (k == "idle")  { m_bg = "#EEF2FA"; m_fg = "#5A6075"; }
            else                   { m_bg = "#E1EAFF"; m_fg = "#1D4ED8"; }      // tint
        } else {
            if (k == "neutral")    { m_bg = "#EEF2FA"; m_fg = "#434655"; m_border = QColor("#E2E7F1"); }
            else if (k == "white") { m_bg = "#FFFFFF"; m_fg = "#434655"; }
            else if (k == "green") { m_bg = "#E3F6EC"; m_fg = "#0B7A4B"; }
            else                   { m_bg = "#E1EAFF"; m_fg = "#1F4FCF"; }      // blue
        }
        setStyleSheet(QString("color:%1; background:transparent; border:none;").arg(m_fg.name()));
        update();
    }
    bool m_pill;
    QColor m_bg, m_fg, m_border;
};

// ======================================================= mock bars
void drawMockBars(QPainter &p, const QRect &r, const QString &codeIn, const QColor &color)
{
    const QString code = codeIn.isEmpty() ? QString("000000") : codeIn;
    const bool dense = r.width() >= 140;              // small glyphs use fewer, thicker bars
    QVector<int> w = {2, 1, 1, 2, 3, 2};                                    // start
    for (int i = 0; i < code.size(); ++i) {
        int d = code[i].digitValue();
        if (d < 0) d = 0;
        w << (d % 3) + 1 << ((d * 7 + i) % 3) + 1 << ((d + i * 2) % 2) + 1
          << ((d * 5 + 3) % 3) + 1 << ((d + i) % 2) + 1 << ((d * 3 + 1) % 3) + 1
;
        if (dense)
            w << ((d * 2 + i) % 3) + 1 << ((d + i * 3) % 2) + 1 << ((d * 4 + 1) % 3) + 1
              << ((d + 2 * i) % 2) + 1 << ((d * 6 + i) % 3) + 1 << ((d + 1) % 2) + 1;
    }
    w << 2 << 3 << 3 << 1 << 1 << 1 << 2;                                   // stop
    int total = 0;
    for (int v : w) total += v;

    p.save();
    p.setRenderHint(QPainter::Antialiasing, false);
    p.setPen(Qt::NoPen);
    p.setBrush(color);
    const double unit = double(r.width()) / total;
    double x = r.left();
    for (int i = 0; i < w.size(); ++i) {
        const double bw = w[i] * unit;
        if (i % 2 == 0)
            p.drawRect(QRect(qRound(x), r.top(), qMax(1, qRound(x + bw) - qRound(x)), r.height()));
        x += bw;
    }
    p.restore();
}

static QString spaced(const QString &s)
{
    QStringList parts;
    for (QChar c : s) parts << QString(c);
    return parts.join("  ");
}

void BarcodeView::paintBarcodeCard(QPainter &p, const QRect &r) const
{
    p.fillRect(r, Qt::white);
    if (!m_image.isNull()) {
        const QPixmap s = m_image.scaled(r.size() - QSize(40, 24), Qt::KeepAspectRatio, Qt::SmoothTransformation);
        p.drawPixmap(r.center().x() - s.width() / 2, r.center().y() - s.height() / 2, s);
        return;
    }
    const int mx = qRound(r.width() * 0.10);
    const int barH = m_digits ? r.height() - 90 : r.height() - 56;
    drawMockBars(p, QRect(r.left() + mx, r.top() + 28, r.width() - 2 * mx, barH), m_code, QColor("#0B1C30"));
    if (m_digits) {
        QFont f = monoFont(22, true);
        f.setLetterSpacing(QFont::AbsoluteSpacing, 0);
        p.setFont(f);
        p.setPen(QColor("#0B1C30"));
        p.drawText(QRect(r.left(), r.top() + 28 + barH + 8, r.width(), 36), Qt::AlignCenter, spaced(m_code));
    }
}

QPixmap BarcodeView::renderBarcode() const
{
    if (!m_image.isNull()) return m_image;          // the backend image, full size and unscaled (so it can be scanned again)
    const QRect base(0, 0, 450, 180);
    QPixmap pm(base.size() * 2);
    pm.fill(Qt::white);
    QPainter p(&pm);
    p.setRenderHint(QPainter::TextAntialiasing);
    p.scale(2, 2);
    paintBarcodeCard(p, base);
    return pm;
}

static int chipWidth(const QFontMetrics &fm, const QString &text, bool dot, bool icon)
{
    return fm.horizontalAdvance(text) + 20 + (dot ? 14 : 0) + (icon ? 20 : 0);
}

static void drawChip(QPainter &p, const QRect &r, const QString &text, const QColor &bg, const QColor &fg,
                     const QColor &dot = QColor(), const QString &icon = QString())
{
    p.setPen(Qt::NoPen);
    p.setBrush(bg);
    p.drawRoundedRect(r, 6, 6);
    int x = r.left() + 10;
    if (dot.isValid()) { p.setBrush(dot); p.drawEllipse(QPoint(x + 3, r.center().y()), 4, 4); x += 14; }
    if (!icon.isEmpty()) { p.drawPixmap(x, r.center().y() - 7, iconPixmap(icon, fg, 14)); x += 20; }
    p.setPen(fg);
    p.drawText(QRect(x, r.top(), r.right() - x, r.height()), Qt::AlignVCenter | Qt::AlignLeft, text);
}

void CameraView::paintEvent(QPaintEvent *)
{
    QPainter p(this);
    p.setRenderHint(QPainter::Antialiasing);
    QPainterPath clip;
    clip.addRoundedRect(QRectF(rect()), 10, 10);
    p.setClipPath(clip);

    QRadialGradient bg(width() / 2.0, height() / 2.0, qMax(width(), height()) * 0.7);
    bg.setColorAt(0, QColor("#10243F"));
    bg.setColorAt(1, QColor("#070F1C"));
    p.fillRect(rect(), bg);

    const bool live = m_running && !m_frame.isNull();
    qreal scale = 1.0;                                                    // widget px per camera px
    if (live) {
        p.setRenderHint(QPainter::SmoothPixmapTransform);
        scale = qMin(qreal(width()) / m_frame.width(), qreal(height()) / m_frame.height());   // show the entire camera frame
        const QSizeF sz(m_frame.width() * scale, m_frame.height() * scale);
        p.drawPixmap(QRectF((width() - sz.width()) / 2, (height() - sz.height()) / 2, sz.width(), sz.height()),
                     m_frame, QRectF(m_frame.rect()));
        p.fillRect(rect(), QColor(5, 12, 24, 40));                        // slight dim so the HUD stays readable

        if (!m_detRect.isEmpty()) {                                       // detected barcode
            const QSizeF fsz(m_frame.width() * scale, m_frame.height() * scale);
            const QPointF org((width() - fsz.width()) / 2, (height() - fsz.height()) / 2);
            const QRectF d(org.x() + m_detRect.x() * scale, org.y() + m_detRect.y() * scale,
                           m_detRect.width() * scale, m_detRect.height() * scale);
            p.setBrush(QColor(52, 211, 153, 28));
            p.setPen(QPen(QColor("#34D399"), 3));
            p.drawRoundedRect(d, 4, 4);
            if (!m_detLabel.isEmpty()) {
                QFont lf = monoFont(14, true);
                p.setFont(lf);
                const QFontMetrics lfm(lf);
                const int lw = lfm.horizontalAdvance(m_detLabel) + 20;
                const QRect lr(int(d.left()), int(d.top()) - 30 < 6 ? int(d.bottom()) + 6 : int(d.top()) - 30, lw, 26);
                p.setPen(Qt::NoPen);
                p.setBrush(QColor("#34D399"));
                p.drawRoundedRect(lr, 6, 6);
                p.setPen(QColor("#052E1C"));
                p.drawText(lr, Qt::AlignCenter, m_detLabel);
            }
        }
    } else {
        p.setPen(QPen(QColor(255, 255, 255, 14), 1.5, Qt::SolidLine, Qt::RoundCap));
        for (int y = 14; y < height(); y += 18)
            for (int x = 14; x < width(); x += 18) p.drawPoint(x, y);
    }

    // scan frame (corner brackets)
    const int fw = int(width() * 0.74), fh = int(height() * 0.64);
    const QRect f((width() - fw) / 2, (height() - fh) / 2, fw, fh);
    p.setBrush(Qt::NoBrush);
    p.setPen(QPen(QColor(m_running ? "#2F6BF0" : "#2A3E63"), 4, Qt::SolidLine, Qt::FlatCap, Qt::MiterJoin));
    const int L = 30;
    auto corner = [&](QPoint c, int dx, int dy) {
        p.drawLine(c, c + QPoint(dx * L, 0));
        p.drawLine(c, c + QPoint(0, dy * L));
    };
    corner(f.topLeft() + QPoint(2, 2), 1, 1);
    corner(f.topRight() + QPoint(-2, 2), -1, 1);
    corner(f.bottomLeft() + QPoint(2, -2), 1, -1);
    corner(f.bottomRight() + QPoint(-2, -2), -1, -1);

    p.setFont(monoFont(11));
    const QFontMetrics fm(p.font());

    if (!m_running) {                                                     // camera off
        const QString t = "CAMERA OFF";
        drawChip(p, QRect(14, 14, chipWidth(fm, t, true, false), 26), t, QColor(0, 0, 0, 110), QColor("#CBD5E1"), QColor("#94A3B8"));
        QFont tf; tf.setPixelSize(17); tf.setWeight(QFont::DemiBold);
        p.setFont(tf);
        p.setPen(QColor("#D5DEEE"));
        p.drawText(f.adjusted(0, 0, 0, -20), Qt::AlignCenter, tr("Camera is off"));
        tf.setPixelSize(13); tf.setWeight(QFont::Normal);
        p.setFont(tf);
        p.setPen(QColor("#8493AE"));
        p.drawText(f.adjusted(0, 34, 0, 0), Qt::AlignCenter, tr("Press \"Start Camera\" to begin scanning"));
        return;
    }

    // scan line
    const qreal t = m_frozen ? 0.5 : (qSin(m_phase) + 1) / 2;
    const int y = f.top() + 14 + int(t * (f.height() - 28));
    QLinearGradient glow(0, y - 26, 0, y + 26);
    glow.setColorAt(0, QColor(34, 211, 238, 0));
    glow.setColorAt(0.5, QColor(34, 211, 238, 55));
    glow.setColorAt(1, QColor(34, 211, 238, 0));
    p.fillRect(QRect(f.left(), y - 26, f.width(), 52), glow);
    QLinearGradient ln(f.left(), 0, f.right(), 0);
    ln.setColorAt(0, QColor(34, 211, 238, 0));
    ln.setColorAt(0.5, QColor(103, 232, 249, 255));
    ln.setColorAt(1, QColor(34, 211, 238, 0));
    p.setPen(QPen(QBrush(ln), 2));
    p.drawLine(f.left(), y, f.right(), y);

    // HUD chips (real values from the camera frame)
    p.setFont(monoFont(11));
    QString left = m_frozen ? QString("FROZEN") : QString("LIVE");
    if (live) left += QString(" SENSOR: %1x%2").arg(m_frame.width()).arg(m_frame.height());
    else      left = "CONNECTING...";
    drawChip(p, QRect(14, 14, chipWidth(fm, left, true, false), 26), left, QColor(0, 0, 0, 120), QColor("#E2E8F0"),
             m_frozen ? QColor("#FBBF24") : QColor("#34D399"));
    if (live) {
        const QString roi = QString("ROI: [%1x%2]").arg(qRound(f.width() / scale)).arg(qRound(f.height() / scale));
        const int rw = chipWidth(fm, roi, false, true);
        drawChip(p, QRect(width() - rw - 14, 14, rw, 26), roi, QColor(0, 0, 0, 120), QColor("#CFE0FF"), QColor(), "target");
    }
}

# Increment 1 — Barcode Generator and Reader

---

## Overview

The first increment focuses on developing a working version of the C++ desktop application with barcode generation and reading capabilities.

The application must be able to generate a barcode from a six-digit number, display it on the screen, save it as an image, and read the generated barcode through either an uploaded image or a laptop camera.

---

## Requirements

### 1. Barcode Generation

The application must:

- Accept a six-digit numeric input from the user.
- Ensure that the first digit is non-zero.
- Generate a barcode corresponding to the provided number.
- Display the generated barcode on the application screen.
- Save the generated barcode to disk as an image.

### 2. Barcode Reading

The application must be able to read a barcode generated through the application.

**A. Image Upload**

- Allow the user to select and upload a barcode image.
- Read and decode the barcode from the selected image.
- Display the decoded six-digit number on the screen.

**B. Laptop Camera**

- Allow the user to scan a barcode using the laptop camera.
- Detect and decode the barcode from the camera feed.
- Display the decoded six-digit number on the screen.

---

## Technical Constraints

- Existing non-standard libraries must not be used for barcode generation.
- Libraries for image processing, file handling, and other supporting operations are permitted.
- The application must be developed in C++.
- All submissions must be made through GitHub.

            '-DBUILD_opencv_python3=OFF','-DWITH_FFMPEG=OFF','-DWITH_MSMF=OFF',
            '-DWITH_DSHOW=ON','-DWITH_IPP=OFF','-DWITH_OPENCL=OFF','-DWITH_OPENEXR=OFF',
            '-DWITH_WEBP=OFF','-DWITH_AVIF=OFF','-DWITH_JASPER=OFF','-DWITH_TIFF=OFF',
            '-DWITH_OPENJPEG=OFF','-DWITH_PROTOBUF=OFF','-DWITH_FLATBUFFERS=OFF','-DCPU_DISPATCH=')
        Run $cmake @('--build',$opencvBuild,'--parallel',"$Jobs")
        Run $cmake @('--install',$opencvBuild)
    }
    Run $cmake @('-S',$repo,'-B',$build,'-G','Ninja','-DCMAKE_BUILD_TYPE=Release',
        "-DCMAKE_CXX_COMPILER=$MinGWRoot/bin/g++.exe","-DCMAKE_MAKE_PROGRAM=$ninja",
        "-DCMAKE_PREFIX_PATH=$QtRoot","-DOpenCV_DIR=$opencv",'-DBUILD_TESTING=ON')
    Run $cmake @('--build',$build,'--parallel',"$Jobs")
    Run $deploy @('--release','--no-translations','--no-system-d3d-compiler','--no-opengl-sw',
        '--compiler-runtime',"$build/BarcodeHub.exe")
    Copy-Item (Join-Path $QtRoot 'plugins/platforms/qoffscreen.dll') (Join-Path $build 'platforms/qoffscreen.dll') -Force
    if (!$SkipTests) {
        Run (Join-Path (Split-Path $cmake -Parent) 'ctest.exe') @('--test-dir',$build,'--output-on-failure')
    }
    $builtExe = Join-Path $build 'BarcodeHub.exe'
    $portableExe = Join-Path $package 'BarcodeHub.exe'
    try {
        Run $cmake @('-E','copy_if_different',$builtExe,$portableExe)
    } catch {
        # Windows permits renaming an executing image. Keep that session intact
        # and retain its executable while publishing the new build for next launch.
        if (!(Test-Path $portableExe)) { throw }
        $previousExe = Join-Path $package ("BarcodeHub.previous-" + [guid]::NewGuid().ToString('N') + '.exe')
        Move-Item -LiteralPath $portableExe -Destination $previousExe
        Copy-Item -LiteralPath $builtExe -Destination $portableExe
        Write-Host "Running session preserved at $previousExe; restart BarcodeHub to use the new build."
    }
    foreach ($dll in Get-ChildItem $build -Filter '*.dll') {
        Run $cmake @('-E','copy_if_different',$dll.FullName,(Join-Path $package $dll.Name))
    }
    Run $deploy @('--release','--no-translations','--no-system-d3d-compiler','--no-opengl-sw',
        '--compiler-runtime',"$package/BarcodeHub.exe")
    $licenses = Join-Path $package 'licenses'
    New-Item -ItemType Directory -Force $licenses | Out-Null
    Copy-Item (Join-Path $repo 'LICENSE') (Join-Path $licenses 'BarcodeHub-LICENSE.txt') -Force
    Copy-Item (Join-Path $opencv 'LICENSE') (Join-Path $licenses 'OpenCV-LICENSE.txt') -Force
    Get-ChildItem (Join-Path $opencv 'etc/licenses') -File -Recurse -ErrorAction SilentlyContinue |
        Copy-Item -Destination $licenses -Force
    $qtLicenses = Join-Path $qtBase 'Licenses'
    if (Test-Path $qtLicenses) {
        $qtLicenseTarget = Join-Path $licenses 'Qt'
        New-Item -ItemType Directory -Force $qtLicenseTarget | Out-Null
        Get-ChildItem $qtLicenses -File | Copy-Item -Destination $qtLicenseTarget -Force
    }
    Write-Host "Ready: $package/BarcodeHub.exe"
} finally {
    $env:PATH = $originalPath
}

#include "barcode/BarcodeGenerator.h"
#include "barcode/Decoder.h"
#include "scanner/ImageScanner.h"
#include "scanner/CameraScanner.h"
#include "scanner/Scanner.h"

#include "decodepage.h"

#include <QApplication>
#include <QElapsedTimer>
#include <QImageReader>
#include <QPainter>
#include <QTableWidget>
#include <QTemporaryDir>
#include <QThread>
#include <QTransform>
#include <iostream>
#include <functional>

static int failures = 0;
static void check(bool ok, const QString &name) {
    std::cout << (ok ? "PASS " : "FAIL ") << name.toStdString() << std::endl;
    if (!ok) ++failures;
}

static void roundTrip(const QImage &image, const QString &code, const QString &name) {
    const auto r = ImageScanner().scan(image);
    check(r.success && r.text == code, name + (r.success ? "" : ": " + r.error));
}

static bool waitFor(const std::function<bool()> &condition, int timeoutMs = 10000) {
    QElapsedTimer timer;
    timer.start();
    while (!condition() && timer.elapsed() < timeoutMs) {
        QApplication::processEvents();
        QThread::msleep(10);
    }
    return condition();
}

static int cameraSmoke() {
    const auto names = CameraScanner::availableCameras();
    check(!names.isEmpty(), "camera enumeration");
    for (const auto &name : names) std::cout << "DEVICE " << name.toStdString() << std::endl;
    if (names.isEmpty()) return 1;
    CameraScanner camera;
    for (int cycle = 0; cycle < 2; ++cycle) {
        QString error;
        const bool opened = camera.openCamera(0, &error);
        check(opened, "open camera cycle " + QString::number(cycle) + ": " + error);
        if (!opened) continue;
        int frames = 0;
        for (int i = 0; i < 20; ++i) {
            const auto frame = camera.readFrame();
            if (frame.success && !frame.frame.isNull()) ++frames;
        }
        check(frames == 20, "20 real frames with decoder enabled");
        camera.closeCamera();
        check(!camera.isOpened(), "camera released");
    }
    {
        MainWindow window;
        auto *decode = window.decodePage();
        decode->setCameraActive(true);
        decode->cameraStartRequested(0);
        check(waitFor([&] { return !decode->cameraView()->frame().isNull(); }), "threaded UI receives real camera frames");
        decode->cameraView()->setFrozen(true);
        const auto frozen = decode->cameraView()->frame().toImage();
        QElapsedTimer delay; delay.start();
        waitFor([&] { return delay.elapsed() > 400; }, 1000);
        check(frozen == decode->cameraView()->frame().toImage(), "live freeze preserves frame");
        decode->stopCamera();
        check(!decode->cameraActive() && decode->cameraView()->frame().isNull(), "UI stop clears camera state");
        // Stop immediately while the worker is opening: late callbacks must not restart it.
        decode->setCameraActive(true);
        decode->cameraStartRequested(0);
        decode->stopCamera();
        delay.restart();
        waitFor([&] { return delay.elapsed() > 1500; }, 2500);
        check(!decode->cameraActive() && decode->cameraView()->frame().isNull(), "cancel opening ignores stale callbacks");
    }
    QString error;
    check(camera.openCamera(0, &error), "window shutdown releases camera for reuse");
    camera.closeCamera();
    return failures ? 1 : 0;
}

int main(int argc, char **argv) {
    QApplication app(argc, argv);
    if (app.arguments().contains("--camera-smoke")) return cameraSmoke();
    BarcodeGenerator generator;
    for (const QString &bad : {QString(), QString("012345"), QString("12345"), QString("1234567"), QString("12a456")})
        check(!generator.generate(bad).success, "reject invalid input: " + bad);
    check(BarcodeGenerator::ean8Payload("123456") == "01234565", "known EAN-8 check digit");

    for (const QString &code : {QString("100000"), QString("123456"), QString("749205"), QString("999999")}) {
        for (bool digits : {false, true}) {
            const auto result = generator.generate(code, digits);
            check(result.success, "generate " + code);
            roundTrip(result.image, code, "round trip " + code + (digits ? " with text" : " without text"));
        }
    }
    const QImage original = generator.generate("749205").image;
    roundTrip(original.mirrored(true, false), "749205", "mirrored barcode");
    roundTrip(original.transformed(QTransform().rotate(90)), "749205", "vertical barcode");
    roundTrip(original.transformed(QTransform().rotate(180)), "749205", "upside down barcode");
    roundTrip(original.scaled(300, 120, Qt::IgnoreAspectRatio, Qt::SmoothTransformation), "749205", "small barcode");
    roundTrip(original.scaled(1800, 720), "749205", "large barcode");
    QImage tilted(900, 500, QImage::Format_RGB32);
    tilted.fill(Qt::white);
    { QPainter p(&tilted); p.translate(450, 250); p.rotate(12); p.drawImage(-300, -120, original); }
    roundTrip(tilted, "749205", "12 degree tilted barcode");
    QImage transparent = original.convertToFormat(QImage::Format_ARGB32);
    for (int y = 0; y < transparent.height(); ++y)
        for (int x = 0; x < transparent.width(); ++x)
            if (qGray(transparent.pixel(x, y)) > 245) transparent.setPixel(x, y, qRgba(0, 0, 0, 0));
    roundTrip(transparent, "749205", "transparent background");
    QTemporaryDir dir;
    const QString path = dir.filePath(QString::fromUtf8("barcode_\xE6\xB5\x8B\xE8\xAF\x95.png"));
    check(original.save(path), "save PNG at Unicode path");
    roundTrip(QImage(path), "749205", "reload saved PNG");
    const QString jpeg = dir.filePath("barcode.jpg");
    check(original.save(jpeg, "JPG", 75), "save JPEG");
    roundTrip(QImage(jpeg), "749205", "JPEG decode");
    QImage blank(600, 240, QImage::Format_RGB32); blank.fill(Qt::white);
    check(!ImageScanner().scan(blank).success, "reject blank image");
    check(!ImageScanner().scan(QImage()).success, "reject empty image");
    QImage damaged = original;
    { QPainter p(&damaged); p.fillRect(250, 0, 30, damaged.height(), Qt::white); }
    check(!ImageScanner().scan(damaged).success, "reject damaged bars");
    // Replace only the check digit with another valid pattern: structure still passes.
    QImage wrongCheck = generator.generate("123456", false).image;
    { QPainter p(&wrongCheck); p.fillRect(464, 26, 49, 188, Qt::white);
      const QString bits = "1110010"; // right-hand encoding of zero, instead of five
      for (int i = 0; i < 7; ++i) if (bits[i] == '1') p.fillRect(464 + 7*i, 26, 7, 188, Qt::black); }
    check(!ImageScanner().scan(wrongCheck).success, "reject wrong checksum");
    std::string stale = "123456";
    Decoder decoder;
    check(!decoder.decodeSix("bad", stale) && stale.empty(), "failed decode clears previous output");

    // Exercise the actual frontend-to-backend signal connections.
    {
        MainWindow window;
        auto *decode = window.decodePage();
        auto *history = decode->findChild<QTableWidget *>("scanHistory");
        check(history && history->rowCount() == 0, "history starts empty");
        window.generatePage()->setCode("749205");
        auto *view = window.generatePage()->findChild<BarcodeView *>();
        roundTrip(view->renderBarcode().toImage(), "749205", "UI generated image decodes");
        decode->sampleRequested("749205", false);
        check(history->rowCount() == 1 && history->item(0, 2)->text() == "749205", "sample uses real decoder and logs result");
        decode->sampleRequested("749205", true);
        check(history->rowCount() == 1, "failed sample does not add success history");
        decode->setTab(2);
        for (auto *tabs : decode->findChildren<SegBar *>()) check(tabs->current() == 2, "tabs stay synchronized");
        decode->setCameraList({});
        check(!decode->cameraActive(), "no camera is inactive");
        CameraView camera;
        camera.setRunning(true);
        camera.setFrame(QPixmap::fromImage(original));
        camera.setFrozen(true);
        camera.setFrame(QPixmap::fromImage(blank));
        check(camera.frame().toImage() == original, "freeze retains displayed frame");
        camera.setRunning(false);
        check(camera.frame().isNull() && !camera.frozen(), "stop clears frame and frozen state");
    }
    std::cout << "Failures: " << failures << std::endl;
    return failures ? 1 : 0;
}

# Contributing Guidelines

This document explains how team members can contribute to **Bar-Code-System-CPP**.

## 1. Clone the Repository

Clone the repository and navigate into the project directory:

```bash
git clone https://github.com/Syntrojex/Bar-Code-System-CPP.git
cd Bar-Code-System-CPP
```

## 2. Switch to Your Assigned Branch

Switch to your assigned branch. Replace `your-branch` with your actual branch name.

```bash
git checkout your-branch
```

Available team branches:

- `ali`
- `farzam`
- `syntrojex`
- `shauzab`

## 3. Make Your Changes

Add or modify the files required for your assigned task.

Follow the existing project structure and coding conventions. Avoid modifying unrelated files.

## 4. Stage and Commit Changes

Check your changes:

```bash
git status
```

Stage your files:

```bash
git add .
```

Commit your changes with a meaningful message:

```bash
git commit -m "Add barcode generation"
```

## 5. Push Your Branch

Push your changes to your assigned branch:

```bash
git push origin your-branch
```

## 6. Create a Pull Request

Submit your changes for integration into `main`.

1. Open the project repository on GitHub.
2. Select **Compare & pull request**.
3. Set the base branch to `main`.
4. Select your assigned branch as the compare branch.
5. Add a meaningful title and description.
6. Create the Pull Request.

## 7. Review and Merge

The project lead will review submitted Pull Requests.

- The project lead's approval is required before merging.
- Address any requested changes before approval.
- Team members should not directly push changes to `main`.
- Only reviewed and approved contributions should be merged.

## 8. Automatic Branch Synchronization

Whenever changes are merged into `main`, GitHub Actions automatically attempts to synchronize all team branches with the latest `main`.

- No manual fetching or merging is required after successful synchronization.
- Existing branch commits and contributions are preserved.
- If a merge conflict occurs, the affected branch will be skipped and must be resolved manually.
- Check the GitHub Actions workflow status if synchronization fails.

## General Guidelines

- Always work on your assigned branch.
- Keep commits meaningful and focused.
- Follow the existing C++ coding conventions.
- Test your changes before submitting a Pull Request.
- Keep Pull Requests limited to the assigned task.
- Coordinate with the project lead when necessary.
- Never commit directly to `main`.

Thank you for contributing to the project.

# Bar-Code-System-CPP

## Run on Windows

Double-click **`Run-BarcodeHub.cmd`**, or launch
**`build/portable/BarcodeHub/BarcodeHub.exe`** after building.
The portable directory includes Qt plugins, OpenCV and MinGW runtime DLLs.
Copy the **whole directory**, not just the executable, to another Windows x64 PC.

### Build and install dependencies

Install the **Qt 6 MinGW 64-bit kit**, its matching MinGW compiler, CMake and Ninja
through the Qt installer. Then run from this repository in PowerShell:

```powershell
powershell -ExecutionPolicy Bypass -File scripts/build-windows.ps1
```

The script detects Qt under `C:\Qt` or `D:\Qt`, chooses its matching compiler,
downloads OpenCV **4.13.0** from the official OpenCV GitHub tag if sources are
missing, builds and installs it locally under `build/deps/opencv`, builds the
application, runs regression tests and creates the portable directory.
It does not require administrator access or change the system PATH.
The first OpenCV build takes several minutes. Later builds reuse it.

For another Qt installation, pass explicit paths:

```powershell
.\scripts\build-windows.ps1 -QtRoot C:\Qt\6.11.0\mingw_64 -MinGWRoot C:\Qt\Tools\mingw1310_64
```

Do not mix MSVC OpenCV binaries with a MinGW Qt kit. OpenCV is required: a build
without working image/camera support now fails at configuration time.
The setup script applies a narrow OpenCV 4.13 MinGW compatibility fix, replacing
the non-standard `M_PI` constant with OpenCV's `CV_PI` in `shapedescr.cpp`.

### Camera and scanning

- Open **Decode Barcode → Scan with Camera**, select a detected camera and
  requested resolution, then click **Start Camera**. The actual frame size is
  shown in the preview; cameras may negotiate a different resolution.
- **Freeze / Resume** freezes both the preview and the decoded result.
  **Mirror** changes only the preview. **Snapshot** saves the displayed frame.
- **Autofocus** requests hardware autofocus where supported. Fixed-focus
  webcams need suitable distance, sharp bars and good lighting.
- Connect a camera and use the refresh button if the device list is empty.
  Stop the camera before changing devices or resolution. Leaving the camera
  tab or closing the application releases it.
- Successful scans appear in **Scan Logs**. CSV export and clear history work;
  logs are held for this session (up to 1,000 entries). Automatic clipboard copy
  is optional and starts off.
- Upload a PNG/JPEG/BMP image or use the built-in clean/damaged samples to
  check decoding without a camera.

This increment supports **the project's EAN-8 format**: a nonzero-leading
six-digit code encoded as `0 + code + Mod-10 check digit`. It does not claim QR,
Code 128, general retail EAN-13, or arbitrary EAN-8 support. SVG export and
thermal printing remain disabled. Printed-camera decoding depends on focus,
barcode size, lighting and camera quality.

If opening fails, close other camera apps and check Windows **Settings →
Privacy & security → Camera → Let desktop apps access your camera**. Reconnect
the webcam and refresh. This application uses OpenCV's DirectShow backend on
Windows and processes frames locally.

### Verify

```powershell
ctest --test-dir build/local-release --output-on-failure
.\build\local-release\barcode_tests.exe --camera-smoke
```

Regression tests cover generation/decode round trips, saved PNG/JPEG images,
Unicode paths, rotation/mirroring, transparent backgrounds, malformed images,
checksum rejection and frontend integration. The separate camera smoke test
opens the first real camera twice, reads frames with decoding enabled and
checks release/reopen; it requires an attached camera.

See [OpenCV build options](https://docs.opencv.org/4.13.0/db/d05/tutorial_config_reference.html)
and [OpenCV source release](https://github.com/opencv/opencv/tree/4.13.0).

A C++ desktop application designed to simplify inventory management and sales operations for small retail stores, supermarkets, and corner shops.

---

## Project Overview

Bar-Code-System-CPP is an academic term project developed for the **Introduction to Software Engineering (ISE)** course at FAST-NUCES.

The application aims to provide a simple and efficient system through which store employees can manage inventory and record sales with minimal manual input.

Cashiers will be able to scan product barcodes, retrieve product information, add items to orders, and generate bills. Managers will also be able to generate barcodes for products in the store.

The project is being developed incrementally, with a focus on building a reliable, user-friendly, and maintainable desktop application.

---

## Key Features

- **Barcode Generation** — Generate barcodes for products.
- **Barcode Scanning** — Read barcodes using a laptop camera.
- **Image-Based Reading** — Upload barcode images for decoding.

---

- **Qt** — Desktop graphical user interface.
